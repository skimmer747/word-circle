```ts
import type { GridPuzzle } from './types';

/** A puzzle, read. Squares are numbered row by row across the whole box: row × size + column. */
export interface GridShape {
  size: number;
  /** True for each square that holds a letter. */
  open: boolean[];
  /** The open squares, in reading order. */
  squares: number[];
  /** Every run of two or more open squares in a line: across (top to bottom), then down (left to right). */
  runs: number[][];
  /** Each square's section (an index into `sections`), or -1 for a blocked square. */
  sectionOf: number[];
  /** Each section's squares, in reading order. */
  sections: number[][];
  /** Each section's letters, in alphabetical order. */
  letters: string[][];
  /** The answer letter in each square ('' for a blocked square). */
  answer: string[];
}

/** Runs of two or more open squares: across first (row by row), then down (column by column). */
export function findRuns(open: readonly boolean[], size: number): number[][] {
  const runs: number[][] = [];
  const scan = (squareAt: (line: number, i: number) => number) => {
    for (let line = 0; line < size; line++) {
      let run: number[] = [];
      for (let i = 0; i <= size; i++) {
        const square = i < size ? squareAt(line, i) : -1;
        if (square >= 0 && open[square]) {
          run.push(square);
        } else {
          if (run.length >= 2) runs.push(run);
          run = [];
        }
      }
    }
  };
  scan((row, col) => row * size + col);
  scan((col, row) => row * size + col);
  return runs;
}

export function readPuzzle(puzzle: GridPuzzle): GridShape {
  const { size, cells } = puzzle;
  const open = [...cells].map((ch) => ch !== '.');
  const sectionOf = [...puzzle.sections].map((code) => (code === '.' ? -1 : code.charCodeAt(0) - 97));
  const sections: number[][] = Array.from({ length: Math.max(-1, ...sectionOf) + 1 }, () => []);
  sectionOf.forEach((section, square) => {
    if (section >= 0) sections[section].push(square);
  });
  return {
    size,
    open,
    squares: open.flatMap((isOpen, square) => (isOpen ? [square] : [])),
    runs: findRuns(open, size),
    sectionOf,
    sections,
    letters: sections.map((squares) => squares.map((square) => cells[square]).sort()),
    answer: [...cells].map((ch) => (ch === '.' ? '' : ch)),
  };
}
```

```ts
import { findRuns, readPuzzle } from '../src/grid/puzzle';
import type { GridPuzzle } from '../src/grid/types';
import { shuffle } from '../src/shared/random';

/** Everyday two-letter words that can be in an answer. */
export const TWO_LETTER_ANSWERS = [
  'am', 'an', 'as', 'at', 'be', 'by', 'do', 'go', 'he', 'hi', 'if', 'in', 'is', 'it',
  'me', 'my', 'no', 'of', 'oh', 'on', 'or', 'ox', 'so', 'to', 'up', 'us', 'we',
];
/** Two-letter words that count as real words but are never in an answer. */
export const TWO_LETTER_EXTRAS = ['ad', 'ah', 'ax', 'eh', 'ha', 'ho', 'id', 'la', 'lo', 'ma', 'pa', 'ye'];

export interface GridWordSources {
  /** Common words (SCOWL size ≤ 20, 3–6 letters). */
  common: Iterable<string>;
  /** Every real word (SCOWL size ≤ 60, 3–6 letters). */
  all: Iterable<string>;
  neverUse: ReadonlySet<string>;
  /** Touchy words: never in an answer, but still real words (words/guess-skip.txt). */
  skip: ReadonlySet<string>;
}

export interface GridWordLists {
  /** What the generator fills grids with, sorted. */
  answers: string[];
  /** Every real word, sorted: what the squiggle and the one-answer checks use. */
  words: string[];
}

export function buildGridWordLists({ common, all, neverUse, skip }: GridWordSources): GridWordLists {
  const words = new Set([...all, ...TWO_LETTER_ANSWERS, ...TWO_LETTER_EXTRAS].filter((word) => !neverUse.has(word)));
  const answers = new Set([...common, ...TWO_LETTER_ANSWERS].filter((word) => words.has(word) && !skip.has(word)));
  return { answers: [...answers].sort(), words: [...words].sort() };
}

/** Answers "can this partly filled run still become a word?" for any mix of letters and "." */
export class PatternIndex {
  private readonly patterns = new Set<string>();

  constructor(words: Iterable<string>) {
    for (const word of words) {
      const n = word.length;
      for (let mask = 0; mask < 1 << n; mask++) {
        let pattern = '';
        for (let i = 0; i < n; i++) pattern += mask & (1 << i) ? '.' : word[i];
        this.patterns.add(pattern);
      }
    }
  }

  has(pattern: string): boolean {
    return this.patterns.has(pattern);
  }
}
```

```ts
/** About 70% of a box's squares are open... */
const OPEN_SHARE = 0.7;
/** ...and never fewer than 60% or more than 80%. */
const OPEN_MIN = 0.6;
const OPEN_MAX = 0.8;
/** Filling gives up on a shape after this many steps, and a new shape is tried. */
const FILL_BUDGET = 20_000;
const ALPHABET = 'abcdefghijklmnopqrstuvwxyz';

function openSquares(open: readonly boolean[]): number[] {
  return open.flatMap((isOpen, square) => (isOpen ? [square] : []));
}

function neighbours(square: number, size: number): number[] {
  const row = Math.floor(square / size);
  const col = square % size;
  const result: number[] = [];
  if (row > 0) result.push(square - size);
  if (row < size - 1) result.push(square + size);
  if (col > 0) result.push(square - 1);
  if (col < size - 1) result.push(square + 1);
  return result;
}

/** The squares of `within` that can be reached from `start` side to side. */
function connected(start: number, within: ReadonlySet<number>, size: number): Set<number> {
  const seen = new Set([start]);
  const todo = [start];
  while (todo.length > 0) {
    for (const next of neighbours(todo.pop()!, size)) {
      if (within.has(next) && !seen.has(next)) {
        seen.add(next);
        todo.push(next);
      }
    }
  }
  return seen;
}

/** What's wrong with a crossword shape; empty when it's fine. */
export function shapeProblems(open: readonly boolean[], size: number): string[] {
  const problems: string[] = [];
  const squares = openSquares(open);
  const share = squares.length / (size * size);
  if (share < OPEN_MIN || share > OPEN_MAX) problems.push(`${squares.length} open squares`);
  for (let line = 0; line < size; line++) {
    if (!squares.some((s) => Math.floor(s / size) === line)) problems.push(`row ${line + 1} is empty`);
    if (!squares.some((s) => s % size === line)) problems.push(`column ${line + 1} is empty`);
  }
  if (squares.length > 0 && connected(squares[0], new Set(squares), size).size !== squares.length) {
    problems.push('not all one piece');
  }
  const inWords = new Set(findRuns(open, size).flat());
  for (const square of squares) if (!inWords.has(square)) problems.push(`square ${square} is in no word`);
  return problems;
}

/** A random crossword shape that follows the rules. */
export function randomShape(size: number, random: () => number): boolean[] {
  for (;;) {
    const open = Array.from({ length: size * size }, () => random() < OPEN_SHARE);
    if (shapeProblems(open, size).length === 0) return open;
  }
}

/**
 * Fills a shape square by square with random letters, backing up whenever a run can no longer become an answer
 * word, so that every run is an answer word and no word appears twice. Null if it can't within `budget` steps.
 */
export function fillShape(
  open: readonly boolean[],
  size: number,
  answers: PatternIndex,
  random: () => number,
  budget = FILL_BUDGET,
): string[] | null {
  const runs = findRuns(open, size);
  const squares = openSquares(open);
  const runsThrough = new Map(squares.map((square) => [square, runs.filter((run) => run.includes(square))]));
  const letters = open.map(() => '');
  const pattern = (run: readonly number[]) => run.map((square) => letters[square] || '.').join('');
  let steps = 0;
  const fill = (i: number): boolean => {
    if (++steps > budget) return false;
    if (i === squares.length) return new Set(runs.map(pattern)).size === runs.length;
    const square = squares[i];
    for (const letter of shuffle([...ALPHABET], random)) {
      letters[square] = letter;
      if (runsThrough.get(square)!.every((run) => answers.has(pattern(run))) && fill(i + 1)) return true;
    }
    letters[square] = '';
    return false;
  };
  return fill(0) ? letters : null;
}
```

```ts
/**
 * How many ways (stopping at `limit`) the open squares can be filled with their sections' letters so that every
 * run is a real word. Fills squares in reading order and checks every run through a square as it goes.
 */
export function countSolutions(
  open: readonly boolean[],
  size: number,
  sectionOf: readonly number[],
  answer: readonly string[],
  words: PatternIndex,
  limit = 2,
): number {
  const runs = findRuns(open, size);
  const squares = openSquares(open);
  const runsThrough = new Map(squares.map((square) => [square, runs.filter((run) => run.includes(square))]));
  const pools = new Map<number, Map<string, number>>();
  for (const square of squares) {
    const pool = pools.get(sectionOf[square]) ?? new Map<string, number>();
    pool.set(answer[square], (pool.get(answer[square]) ?? 0) + 1);
    pools.set(sectionOf[square], pool);
  }
  const letters = open.map(() => '');
  const pattern = (run: readonly number[]) => run.map((square) => letters[square] || '.').join('');
  let count = 0;
  const fill = (i: number): boolean => {
    if (i === squares.length) return ++count >= limit;
    const square = squares[i];
    const pool = pools.get(sectionOf[square])!;
    for (const [letter, left] of pool) {
      if (left === 0) continue;
      pool.set(letter, left - 1);
      letters[square] = letter;
      const stop = runsThrough.get(square)!.every((run) => words.has(pattern(run))) && fill(i + 1);
      pool.set(letter, left);
      letters[square] = '';
      if (stop) return true;
    }
    return false;
  };
  fill(0);
  return count;
}

function touching(a: readonly number[], b: readonly number[], size: number): boolean {
  const inB = new Set(b);
  return a.some((square) => neighbours(square, size).some((next) => inB.has(next)));
}

/**
 * Splits a filled shape into sections that leave exactly one answer. Every square starts as its own section (one
 * answer); then neighbouring sections are merged, smallest pairs first, and a merge is kept only if there is still
 * one answer. Merging only ever adds answers, so a merge that failed once is never tried again.
 * Returns each square's section, or -1 for a blocked square.
 */
export function carveSections(
  open: readonly boolean[],
  size: number,
  answer: readonly string[],
  words: PatternIndex,
  maxSection: number,
  random: () => number,
): number[] {
  let groups = openSquares(open).map((square) => [square]);
  const failed = new Set<string>();
  const key = (a: readonly number[], b: readonly number[]) => [...a, ...b].sort((x, y) => x - y).join(',');
  const sectionsOf = (list: readonly number[][]) => {
    const of = open.map(() => -1);
    list.forEach((group, i) => group.forEach((square) => (of[square] = i)));
    return of;
  };
  for (;;) {
    const pairs: Array<[number, number]> = [];
    for (let i = 0; i < groups.length; i++) {
      for (let j = i + 1; j < groups.length; j++) {
        if (groups[i].length + groups[j].length > maxSection || failed.has(key(groups[i], groups[j]))) continue;
        if (touching(groups[i], groups[j], size)) pairs.push([i, j]);
      }
    }
    if (pairs.length === 0) return sectionsOf(groups);
    const sizeOf = ([i, j]: [number, number]) => groups[i].length + groups[j].length;
    const smallest = Math.min(...pairs.map(sizeOf));
    const choices = pairs.filter((pair) => sizeOf(pair) === smallest);
    const [i, j] = choices[Math.floor(random() * choices.length)];
    const merged = [...groups[i], ...groups[j]].sort((x, y) => x - y);
    const next = [...groups.filter((_, k) => k !== i && k !== j), merged];
    if (countSolutions(open, size, sectionsOf(next), answer, words) === 1) groups = next;
    else failed.add(key(groups[i], groups[j]));
  }
}

/** A puzzle as stored: sections lettered a, b, c… in order of first appearance. */
export function encodePuzzle(size: number, answer: readonly string[], sectionOf: readonly number[]): GridPuzzle {
  const names = new Map<number, string>();
  const sections = sectionOf
    .map((section) => {
      if (section < 0) return '.';
      if (!names.has(section)) names.set(section, String.fromCharCode(97 + names.size));
      return names.get(section)!;
    })
    .join('');
  return { size, cells: answer.map((letter) => letter || '.').join(''), sections };
}
```

```ts
export interface LevelRules {
  /** Squares across (and down) in the box. */
  size: number;
  /** The most letters in one section. */
  maxSection: number;
}

export interface GridIndexes {
  answers: PatternIndex;
  words: PatternIndex;
}

/** One puzzle for the level, or null if this attempt didn't work out (the caller tries again). */
export function makePuzzle(rules: LevelRules, indexes: GridIndexes, random: () => number): GridPuzzle | null {
  const open = randomShape(rules.size, random);
  const answer = fillShape(open, rules.size, indexes.answers, random);
  if (!answer) return null;
  const sectionOf = carveSections(open, rules.size, answer, indexes.words, rules.maxSection, random);
  const puzzle = encodePuzzle(rules.size, answer, sectionOf);
  // A one-letter section is a free letter; one is fine, more makes it too easy.
  const singles = readPuzzle(puzzle).sections.filter((squares) => squares.length === 1).length;
  return singles <= 1 ? puzzle : null;
}

/** Up to `count` different puzzles for a level. */
export function buildLevel(rules: LevelRules, indexes: GridIndexes, count: number, random: () => number): GridPuzzle[] {
  const puzzles: GridPuzzle[] = [];
  const seen = new Set<string>();
  for (let attempt = 0; attempt < count * 20 && puzzles.length < count; attempt++) {
    const puzzle = makePuzzle(rules, indexes, random);
    if (puzzle && !seen.has(puzzle.cells)) {
      seen.add(puzzle.cells);
      puzzles.push(puzzle);
    }
  }
  return puzzles;
}

export function wordsByLength(words: Iterable<string>): Map<number, string[]> {
  const byLength = new Map<number, string[]>();
  for (const word of words) {
    const list = byLength.get(word.length) ?? [];
    list.push(word);
    byLength.set(word.length, list);
  }
  return byLength;
}

/**
 * Every answer to a puzzle (stopping at `limit`), as cells strings, found a different way from countSolutions:
 * word by word, longest runs first, using up each section's letters. Used to check the puzzles we ship.
 */
export function solveByWords(puzzle: GridPuzzle, byLength: ReadonlyMap<number, readonly string[]>, limit = 2): string[] {
  const shape = readPuzzle(puzzle);
  const runs = [...shape.runs].sort((a, b) => b.length - a.length);
  const pools = shape.letters.map((letters) => {
    const pool = new Map<string, number>();
    for (const letter of letters) pool.set(letter, (pool.get(letter) ?? 0) + 1);
    return pool;
  });
  const letters = shape.open.map(() => '');
  const found: string[] = [];
  const place = (k: number): boolean => {
    if (k === runs.length) {
      found.push(letters.map((letter) => letter || '.').join(''));
      return found.length >= limit;
    }
    const run = runs[k];
    for (const word of byLength.get(run.length) ?? []) {
      const placed: number[] = [];
      let fits = true;
      for (let i = 0; i < run.length && fits; i++) {
        const square = run[i];
        if (letters[square]) {
          fits = letters[square] === word[i];
          continue;
        }
        const pool = pools[shape.sectionOf[square]];
        const left = pool.get(word[i]) ?? 0;
        if (left === 0) {
          fits = false;
          continue;
        }
        pool.set(word[i], left - 1);
        letters[square] = word[i];
        placed.push(square);
      }
      const stop = fits && place(k + 1);
      for (const square of placed) {
        const pool = pools[shape.sectionOf[square]];
        pool.set(letters[square], pool.get(letters[square])! + 1);
        letters[square] = '';
      }
      if (stop) return true;
    }
    return false;
  };
  place(0);
  return found;
}

const sectionName = (i: number) => String.fromCharCode(97 + i);

/** Everything wrong with one puzzle; empty when it's fine. */
export function puzzleProblems(
  puzzle: GridPuzzle,
  rules: LevelRules,
  answers: ReadonlySet<string>,
  byLength: ReadonlyMap<number, readonly string[]>,
): string[] {
  const problems: string[] = [];
  const say = (problem: string) => problems.push(`${puzzle.cells}: ${problem}`);
  const { size } = rules;
  if (puzzle.size !== size || puzzle.cells.length !== size * size || puzzle.sections.length !== size * size) {
    say('is the wrong size');
    return problems;
  }
  if (![...puzzle.cells].every((ch, i) => (ch === '.') === (puzzle.sections[i] === '.'))) {
    say("has blocked squares that don't match its sections");
    return problems;
  }
  const shape = readPuzzle(puzzle);
  for (const problem of shapeProblems(shape.open, size)) say(problem);
  const spelled = shape.runs.map((run) => run.map((square) => shape.answer[square]).join(''));
  for (const word of spelled) if (!answers.has(word)) say(`"${word}" is not an answer word`);
  if (new Set(spelled).size !== spelled.length) say('repeats a word');
  shape.sections.forEach((squares, i) => {
    if (squares.length === 0) say(`section ${sectionName(i)} is empty`);
    else if (connected(squares[0], new Set(squares), size).size !== squares.length) say(`section ${sectionName(i)} is in pieces`);
    if (squares.length > rules.maxSection) say(`section ${sectionName(i)} is too big`);
  });
  if (shape.sections.filter((squares) => squares.length === 1).length > 1) say('has more than one one-letter section');
  const solutions = solveByWords(puzzle, byLength);
  if (solutions.length !== 1) say(`has ${solutions.length} answers`);
  else if (solutions[0] !== puzzle.cells) say("its one answer isn't the stored one");
  return problems;
}

/** Problems with a whole level: repeated puzzles, or too few. */
export function levelProblems(level: string, puzzles: readonly GridPuzzle[], minimum: number): string[] {
  const problems: string[] = [];
  const seen = new Set<string>();
  for (const { cells } of puzzles) {
    if (seen.has(cells)) problems.push(`${level}: ${cells} appears twice`);
    seen.add(cells);
  }
  if (puzzles.length < minimum) problems.push(`${level}: only ${puzzles.length} puzzles could be made (need at least ${minimum})`);
  return problems;
}
```

```ts
import { readPuzzle, type GridShape } from './puzzle';
import type { GridPuzzle } from './types';

/** One of a section's letters, and whether it's already on the grid. */
export interface TrayLetter {
  letter: string;
  used: boolean;
}

const LETTER = /^[a-z]$/;

/** One Word Grid puzzle in play: the letters she has placed and the squares Hint has filled. No screen code. */
export class GridGame {
  readonly shape: GridShape;
  private readonly cells: string[];
  private readonly locked: Set<number>;

  /** `placed` and `hinted` come from her save; anything that doesn't fit the puzzle is dropped. */
  constructor(
    puzzle: GridPuzzle,
    placed: readonly string[],
    hinted: readonly number[],
    private readonly isWord: (word: string) => boolean,
  ) {
    this.shape = readPuzzle(puzzle);
    const { open, answer, squares } = this.shape;
    this.cells = open.map(() => '');
    this.locked = new Set(hinted.filter((square) => open[square] === true));
    for (const square of this.locked) this.cells[square] = answer[square];
    for (const square of squares) {
      const letter = placed[square];
      if (!this.locked.has(square) && typeof letter === 'string' && LETTER.test(letter) && this.room(square, letter)) {
        this.cells[square] = letter;
      }
    }
  }

  /** The letter in each square of the box ('' for none), row by row. */
  get placed(): string[] {
    return [...this.cells];
  }

  /** Squares filled by Hint, in order. */
  get hinted(): number[] {
    return [...this.locked].sort((a, b) => a - b);
  }

  /** Every square filled and every run a real word. Each puzzle has one answer, so this is it. */
  get solved(): boolean {
    return this.shape.squares.every((square) => this.cells[square]) && this.badRuns().length === 0;
  }

  letterAt(square: number): string {
    return this.cells[square] ?? '';
  }

  isOpen(square: number): boolean {
    return this.shape.open[square] === true;
  }

  isHinted(square: number): boolean {
    return this.locked.has(square);
  }

  sectionOf(square: number): number {
    return this.shape.sectionOf[square] ?? -1;
  }

  /** A section's letters in alphabetical order, the ones on the grid marked used (repeated letters counted). */
  tray(section: number): TrayLetter[] {
    const onGrid = new Map<string, number>();
    for (const square of this.shape.sections[section] ?? []) {
      const letter = this.cells[square];
      if (letter) onGrid.set(letter, (onGrid.get(letter) ?? 0) + 1);
    }
    return (this.shape.letters[section] ?? []).map((letter) => {
      const left = onGrid.get(letter) ?? 0;
      if (left > 0) onGrid.set(letter, left - 1);
      return { letter, used: left > 0 };
    });
  }

  /** Puts one of the section's free letters in `square`; whatever was there goes back to the tray. */
  place(square: number, letter: string): boolean {
    if (!this.isOpen(square) || this.locked.has(square) || !this.room(square, letter)) return false;
    this.cells[square] = letter;
    return true;
  }

  /** Moves the letter in `from` to `to` in the same section, swapping with whatever is there. */
  move(from: number, to: number): boolean {
    if (from === to || !this.isOpen(from) || !this.isOpen(to) || !this.cells[from]) return false;
    if (this.locked.has(from) || this.locked.has(to) || this.sectionOf(from) !== this.sectionOf(to)) return false;
    [this.cells[from], this.cells[to]] = [this.cells[to], this.cells[from]];
    return true;
  }

  /** Takes a letter off the grid, back to the tray. */
  remove(square: number): boolean {
    if (!this.isOpen(square) || this.locked.has(square) || !this.cells[square]) return false;
    this.cells[square] = '';
    return true;
  }

  /** The runs (indexes into shape.runs) that are full but aren't real words. */
  badRuns(): number[] {
    return this.shape.runs.flatMap((run, i) => {
      const word = run.map((square) => this.cells[square]).join('');
      return word.length === run.length && !this.isWord(word) ? [i] : [];
    });
  }

  /**
   * Puts the right letter in `selected`, or, if that's already right, in the first square that's empty or wrong,
   * and locks it. Returns that square, or null when everything is right.
   */
  hint(selected: number): number | null {
    const { answer, squares } = this.shape;
    const wrong = (square: number) => this.cells[square] !== answer[square];
    const target = this.isOpen(selected) && wrong(selected) ? selected : squares.find(wrong) ?? null;
    if (target === null) return null;
    const letter = answer[target];
    this.cells[target] = '';
    if (!this.room(target, letter)) {
      // Every copy is on the grid, so one of them is in a square where it's wrong: take it from there.
      const from = this.shape.sections[this.sectionOf(target)].find((s) => this.cells[s] === letter && answer[s] !== letter);
      if (from !== undefined) this.cells[from] = '';
    }
    this.cells[target] = letter;
    this.locked.add(target);
    return target;
  }

  /** The first empty square in reading order, or the first square when the grid is full. */
  firstEmpty(): number {
    return this.shape.squares.find((square) => !this.cells[square]) ?? this.shape.squares[0];
  }

  /** The next empty square of `square`'s section in reading order (wrapping), or null when it's full. */
  nextEmptyInSection(square: number): number | null {
    const squares = this.shape.sections[this.sectionOf(square)] ?? [];
    const at = squares.indexOf(square);
    for (let k = 1; k < squares.length; k++) {
      const next = squares[(at + k) % squares.length];
      if (!this.cells[next]) return next;
    }
    return null;
  }

  /** True if the section has a copy of `letter` that isn't already in one of its other squares. */
  private room(square: number, letter: string): boolean {
    const section = this.sectionOf(square);
    if (section < 0) return false;
    const have = this.shape.letters[section].filter((l) => l === letter).length;
    const used = this.shape.sections[section].filter((s) => s !== square && this.cells[s] === letter).length;
    return used < have;
  }
}
```

```ts
import type { KeyValueStore } from '../shared/store';
import { GRID_LEVELS, type GridLevel, type GridPuzzleSet } from './types';

export const GRID_KEY = 'word-grid:v1';

/** Where she is in one level. */
export interface GridProgress {
  /** Her place in the level's list. */
  index: number;
  /** The puzzle at that place (its cells string), to spot a regenerated list. */
  id: string;
  /** The letter in each square of the box ('' for none), row by row. Empty when she hasn't started. */
  placed: string[];
  /** Squares filled by Hint. */
  hinted: number[];
}

export interface GridStats {
  totalSolved: number;
  todaySolved: number;
  lastPlayedDay: string | null;
}

/** Her level, her place in each level she has played, and her counts. */
export interface SavedGrid {
  level: GridLevel;
  levels: Partial<Record<GridLevel, GridProgress>>;
  stats: GridStats;
}

const DAY = /^\d{4}-\d{2}-\d{2}$/;
const LETTER_OR_EMPTY = /^[a-z]?$/;

export function emptyGridStats(): GridStats {
  return { totalSolved: 0, todaySolved: 0, lastPlayedDay: null };
}

export function freshGrid(): SavedGrid {
  return { level: 'easy', levels: {}, stats: emptyGridStats() };
}

function freshProgress(puzzles: GridPuzzleSet, level: GridLevel, index: number): GridProgress {
  const list = puzzles[level];
  const i = index % list.length;
  return { index: i, id: list[i].cells, placed: [], hinted: [] };
}

/** Her place in `level`: the saved one if that puzzle is still there, otherwise that place, fresh. */
export function progressFor(saved: SavedGrid, puzzles: GridPuzzleSet, level: GridLevel): GridProgress {
  const progress = saved.levels[level];
  if (!progress) return freshProgress(puzzles, level, 0);
  const index = progress.index % puzzles[level].length;
  return puzzles[level][index].cells === progress.id ? { ...progress, index } : freshProgress(puzzles, level, index);
}

/** Moves `level` on to its next puzzle (wrapping at the end), with nothing placed. */
export function nextPuzzle(saved: SavedGrid, puzzles: GridPuzzleSet, level: GridLevel): SavedGrid {
  const next = freshProgress(puzzles, level, progressFor(saved, puzzles, level).index + 1);
  return { ...saved, levels: { ...saved.levels, [level]: next } };
}

/** Counts a finished puzzle; "today" starts over on a new day. */
export function recordGridSolve(stats: GridStats, day: string): GridStats {
  return {
    totalSolved: stats.totalSolved + 1,
    todaySolved: stats.lastPlayedDay === day ? stats.todaySolved + 1 : 1,
    lastPlayedDay: day,
  };
}

const isRecord = (value: unknown): value is Record<string, unknown> => typeof value === 'object' && value !== null;
const isCount = (value: unknown): boolean => Number.isInteger(value) && (value as number) >= 0;

function readStats(value: unknown): GridStats {
  if (
    isRecord(value) &&
    isCount(value.totalSolved) &&
    isCount(value.todaySolved) &&
    (value.lastPlayedDay === null || (typeof value.lastPlayedDay === 'string' && DAY.test(value.lastPlayedDay)))
  ) {
    return {
      totalSolved: value.totalSolved as number,
      todaySolved: value.todaySolved as number,
      lastPlayedDay: value.lastPlayedDay as string | null,
    };
  }
  return emptyGridStats();
}

function readProgress(value: unknown): GridProgress | null {
  if (!isRecord(value)) return null;
  const { index, id, placed, hinted } = value;
  if (!isCount(index) || typeof id !== 'string') return null;
  if (!Array.isArray(placed) || !placed.every((l) => typeof l === 'string' && LETTER_OR_EMPTY.test(l))) return null;
  if (!Array.isArray(hinted) || !hinted.every(isCount)) return null;
  return { index: index as number, id, placed: placed as string[], hinted: hinted as number[] };
}

/** A damaged level is dropped on its own, so it never costs her the others. */
function readSavedGrid(value: unknown): SavedGrid | null {
  if (!isRecord(value) || !GRID_LEVELS.includes(value.level as GridLevel) || !isRecord(value.levels)) return null;
  const levels: SavedGrid['levels'] = {};
  for (const level of GRID_LEVELS) {
    const progress = readProgress(value.levels[level]);
    if (progress) levels[level] = progress;
  }
  return { level: value.level as GridLevel, levels, stats: readStats(value.stats) };
}

export function loadSavedGrid(store: KeyValueStore): SavedGrid {
  try {
    const raw = store.getItem(GRID_KEY);
    const saved = raw === null ? null : readSavedGrid(JSON.parse(raw));
    if (saved) return saved;
  } catch {
    // Unreadable save: start fresh.
  }
  return freshGrid();
}

export function saveSavedGrid(store: KeyValueStore, saved: SavedGrid): void {
  try {
    store.setItem(GRID_KEY, JSON.stringify(saved));
  } catch {
    // Storage full or unavailable: keep playing; progress just won't survive a restart.
  }
}
```

```ts
import type { FrameLayout, Rect } from '../frame/layout';

/** The left column: Easy, Medium, Hard, then Hint. */
const LEFT_WIDTH = 130;
const LEVEL_HEIGHT = 64;
const LEVEL_GAP = 12;
const HINT_GAP = 28;
const HINT_HEIGHT = 96;
/** Space between the left column, the tray, the grid and the top-right buttons. */
const GAP = 16;
/** The tray is this share of the screen height, within the limits below. */
const TRAY_SHARE = 0.12;
const TRAY_MIN = 96;
const TRAY_MAX = 110;
const TRAY_PADDING = 6;
const TILE_GAP = 12;
const SQUARE_MAX = 150;
const LETTER_SCALE = 0.62;
/** A section's letters, printed small in its first square. */
const CORNER_SCALE = 0.18;
const CORNER_MIN = 18;

export interface GridLayout {
  frame: FrameLayout;
  /** Easy, Medium, Hard. */
  levels: Rect[];
  hintButton: Rect;
  /** The yellow backing for the selected section's letters. */
  tray: Rect;
  tileSize: number;
  tileFont: number;
  /** Every square of the puzzle's box, row by row, blocked ones included. */
  squares: Rect[];
  squareSize: number;
  squareFont: number;
  cornerFont: number;
}

export function computeGridLayout(frame: FrameLayout, size: number): GridLayout {
  const { play } = frame;
  const levels = [0, 1, 2].map((i) => ({ x: play.x, y: play.y + i * (LEVEL_HEIGHT + LEVEL_GAP), w: LEFT_WIDTH, h: LEVEL_HEIGHT }));
  const hintButton = { x: play.x, y: levels[2].y + LEVEL_HEIGHT + HINT_GAP, w: LEFT_WIDTH, h: HINT_HEIGHT };

  const left = play.x + LEFT_WIDTH + GAP;
  const trayH = Math.round(Math.min(TRAY_MAX, Math.max(TRAY_MIN, frame.height * TRAY_SHARE)));
  const tray = { x: left, y: play.y, w: frame.corner.x - GAP - left, h: trayH };
  const tileSize = trayH - 2 * TRAY_PADDING;

  const top = tray.y + trayH + GAP;
  const right = play.x + play.w;
  const bottom = play.y + play.h;
  const squareSize = Math.floor(Math.min(SQUARE_MAX, (right - left) / size, (bottom - top) / size));
  const x0 = left + (right - left - size * squareSize) / 2;
  const y0 = top + (bottom - top - size * squareSize) / 2;
  const squares = Array.from({ length: size * size }, (_, i) => ({
    x: x0 + (i % size) * squareSize,
    y: y0 + Math.floor(i / size) * squareSize,
    w: squareSize,
    h: squareSize,
  }));

  return {
    frame,
    levels,
    hintButton,
    tray,
    tileSize,
    tileFont: Math.round(tileSize * LETTER_SCALE),
    squares,
    squareSize,
    squareFont: Math.round(squareSize * LETTER_SCALE),
    cornerFont: Math.max(CORNER_MIN, Math.round(squareSize * CORNER_SCALE)),
  };
}

/** Where the tray's `count` tiles go: side by side, centered in the tray. */
export function trayTiles(layout: GridLayout, count: number): Rect[] {
  const { tray, tileSize } = layout;
  const total = count * tileSize + Math.max(0, count - 1) * TILE_GAP;
  const x0 = tray.x + (tray.w - total) / 2;
  return Array.from({ length: count }, (_, i) => ({ x: x0 + i * (tileSize + TILE_GAP), y: tray.y + TRAY_PADDING, w: tileSize, h: tileSize }));
}
```

```ts
import type { Rect } from '../frame/layout';
import { el, place, restartAnimation, wait } from '../ui/dom';
import { CHECK_ICON, HINT_ICON } from '../ui/icons';
import { trayTiles, type GridLayout } from './layout';
import type { GridShape } from './puzzle';
import type { TrayLetter } from './rules';
import { GRID_LEVELS, type GridLevel } from './types';

export interface GridViewState {
  level: GridLevel;
  shape: GridShape;
  /** The letter in each square of the box ('' for none). */
  letters: readonly string[];
  hinted: ReadonlySet<number>;
  selected: number;
  /** The selected section's letters, for the tray. */
  tray: readonly TrayLetter[];
  /** Every section's letters, printed small in its first square. */
  corners: readonly TrayLetter[][];
  /** The squares of each full run that isn't a real word. */
  badRuns: readonly number[][];
}

const LEVEL_LABELS: Record<GridLevel, string> = { easy: 'Easy', medium: 'Medium', hard: 'Hard' };
/** Each square draws its share of a line: thin inside a section, half of a thick line between sections, all of it on the outside. */
const THIN = 1;
const HALF = 3;
const EDGE = 6;
/** Words turn green one at a time when she finishes. */
const WORD_MS = 250;
/** A letter glides back this long (matches the CSS). */
const GLIDE_MS = 250;
/** A picked-up letter is this much bigger than a tray tile. */
const LIFT = 1.1;
const SVG_NS = 'http://www.w3.org/2000/svg';

interface Cell {
  cell: HTMLElement;
  letter: HTMLElement;
  corner: HTMLElement | null;
}

/** Draws Word Grid and plays its animations. Knows nothing about rules. */
export class GridView {
  private layout!: GridLayout;
  private board: HTMLElement | null = null;
  private readonly cells = new Map<number, Cell>();
  private tiles: HTMLElement[] = [];
  private traySection = -1;
  private squiggles: SVGSVGElement | null = null;
  private shownBad = new Set<string>();
  private ghost: HTMLElement | null = null;
  private ghostTimer = 0;
  private dragSource: string | null = null;
  private hintButton: HTMLElement | null = null;

  /** Buttons and letters are drawn here but touched through the touch tracker, which works with a resting thumb. */
  constructor(private readonly layer: HTMLElement) {}

  render(layout: GridLayout, state: GridViewState, animateIn: boolean): void {
    this.layout = layout;
    this.dragSource = null;
    this.traySection = -1;
    this.shownBad = new Set();
    const board = document.createElement('div');
    board.className = 'board grid';
    this.layer.replaceChildren(board);
    this.board = board;

    GRID_LEVELS.forEach((level, i) => {
      const on = level === state.level;
      const button = el('button', on ? 'level on' : 'level', board);
      button.type = 'button';
      button.innerHTML = `${on ? CHECK_ICON : ''}<span>${LEVEL_LABELS[level]}</span>`;
      place(button, layout.levels[i]);
    });
    const hint = el('button', 'button hint', board);
    hint.type = 'button';
    hint.innerHTML = `${HINT_ICON}<span>Hint</span>`;
    place(hint, layout.hintButton);
    this.hintButton = hint;
    place(el('div', 'tray', board), layout.tray);

    this.cells.clear();
    const { shape } = state;
    const firsts = new Set(shape.sections.map((squares) => squares[0]));
    for (const square of shape.squares) {
      const cell = el('div', 'cell', board);
      place(cell, layout.squares[square]);
      const sides = this.sides(shape, square);
      cell.style.borderWidth = sides.map(([width]) => `${width}px`).join(' ');
      cell.style.borderColor = sides.map(([, color]) => color).join(' ');
      const letter = el('span', 'cell-letter', cell);
      letter.style.fontSize = `${layout.squareFont}px`;
      let corner: HTMLElement | null = null;
      if (firsts.has(square)) {
        corner = el('span', 'cell-corner', cell);
        corner.style.fontSize = `${layout.cornerFont}px`;
      }
      this.cells.set(square, { cell, letter, corner });
    }

    const squiggles = document.createElementNS(SVG_NS, 'svg');
    squiggles.setAttribute('class', 'squiggles');
    board.appendChild(squiggles);
    this.squiggles = squiggles;
    this.tiles = [];
    this.ghost = el('div', 'tile ghost', board);

    this.update(state);
    if (animateIn) {
      board.classList.add('entering');
      void board.offsetWidth;
      board.classList.remove('entering');
    }
  }

  /** Brings every square, the corner letters, the tray and the squiggles up to date. */
  update(state: GridViewState): void {
    const { shape } = state;
    const section = shape.sectionOf[state.selected];
    for (const [square, { cell, letter, corner }] of this.cells) {
      letter.textContent = state.letters[square].toUpperCase();
      cell.classList.toggle('hinted', state.hinted.has(square));
      cell.classList.toggle('in-section', shape.sectionOf[square] === section);
      cell.classList.toggle('selected', square === state.selected);
      cell.classList.toggle('lifted', this.dragSource === `square:${square}`);
      if (corner) {
        corner.innerHTML = state.corners[shape.sectionOf[square]]
          .map(({ letter: l, used }) => `<span${used ? ' class="used"' : ''}>${l.toUpperCase()}</span>`)
          .join('');
      }
    }
    this.renderTray(state.tray, section);
    this.renderSquiggles(state.badRuns);
  }

  /** A letter pops into its square. */
  popSquare(square: number): void {
    const cell = this.cells.get(square)?.cell;
    if (cell) restartAnimation(cell, 'pop');
  }

  setPressed(id: string, on: boolean): void {
    if (id.startsWith('tray:')) this.tiles[Number(id.slice('tray:'.length))]?.classList.toggle('pressed', on);
  }

  setHintPulse(on: boolean): void {
    this.hintButton?.classList.toggle('pulse', on);
  }

  /** She picked up a letter from `source` ("tray:i" or "square:n"): it follows her finger. */
  startDrag(letter: string, source: string): void {
    const ghost = this.ghost;
    if (!ghost) return;
    window.clearTimeout(this.ghostTimer);
    this.dragSource = source;
    const size = this.layout.tileSize * LIFT;
    ghost.textContent = letter.toUpperCase();
    ghost.style.transition = 'none';
    ghost.style.width = `${size}px`;
    ghost.style.height = `${size}px`;
    ghost.style.fontSize = `${Math.round(this.layout.tileFont * LIFT)}px`;
    ghost.classList.add('visible');
    if (source.startsWith('tray:')) this.tiles[Number(source.slice('tray:'.length))]?.classList.add('lifted');
    else this.cells.get(Number(source.slice('square:'.length)))?.cell.classList.add('lifted');
  }

  moveDrag(x: number, y: number): void {
    const ghost = this.ghost;
    if (!ghost) return;
    const size = this.layout.tileSize * LIFT;
    ghost.style.left = `${x - size / 2}px`;
    ghost.style.top = `${y - size / 2}px`;
  }

  /** She let go: the letter glides to `to` (back to the tray or its square), or simply vanishes once placed. */
  dropDrag(to: Rect | null): void {
    const ghost = this.ghost;
    this.dragSource = null;
    for (const tile of this.tiles) tile.classList.remove('lifted');
    for (const { cell } of this.cells.values()) cell.classList.remove('lifted');
    if (!ghost) return;
    if (!to) {
      ghost.classList.remove('visible');
      return;
    }
    ghost.style.transition = '';
    place(ghost, to);
    this.ghostTimer = window.setTimeout(() => ghost.classList.remove('visible'), GLIDE_MS);
  }

  /** The words turn green one at a time. */
  async finish(runs: readonly number[][]): Promise<void> {
    for (const run of runs) {
      for (const square of run) this.cells.get(square)?.cell.classList.add('done');
      await wait(WORD_MS);
    }
  }

  private renderTray(tray: readonly TrayLetter[], section: number): void {
    const board = this.board;
    if (!board) return;
    for (const tile of this.tiles) tile.remove();
    const fresh = section !== this.traySection;
    this.traySection = section;
    this.tiles = trayTiles(this.layout, tray.length).map((rect, i) => {
      const tile = el('div', tray[i].used ? 'tile used' : 'tile', board);
      tile.textContent = tray[i].letter.toUpperCase();
      tile.style.fontSize = `${this.layout.tileFont}px`;
      if (fresh) tile.classList.add('appear');
      if (this.dragSource === `tray:${i}`) tile.classList.add('lifted');
      place(tile, rect);
      return tile;
    });
    // The picked-up letter stays above the tray.
    if (this.ghost) board.appendChild(this.ghost);
  }

  /** A soft wavy line through each full run that isn't a word; new ones fade in. */
  private renderSquiggles(runs: readonly number[][]): void {
    const svg = this.squiggles;
    if (!svg) return;
    const keys = runs.map((run) => run.join(','));
    svg.replaceChildren(
      ...runs.map((run, i) => {
        const path = document.createElementNS(SVG_NS, 'path');
        path.setAttribute('d', this.wave(run));
        path.setAttribute('class', this.shownBad.has(keys[i]) ? 'squiggle' : 'squiggle appear');
        return path;
      }),
    );
    this.shownBad = new Set(keys);
  }

  private wave(run: readonly number[]): string {
    const size = this.layout.squareSize;
    const first = this.layout.squares[run[0]];
    const last = this.layout.squares[run[run.length - 1]];
    const across = first.y === last.y;
    const inset = size * 0.15;
    const length = (across ? last.x - first.x : last.y - first.y) + size - 2 * inset;
    const steps = Math.max(2, Math.round(length / (size * 0.25)));
    const step = length / steps;
    const bend = size * 0.2;
    let d = across ? `M ${first.x + inset} ${first.y + size / 2}` : `M ${first.x + size / 2} ${first.y + inset}`;
    for (let i = 0; i < steps; i++) {
      const out = i % 2 === 0 ? -bend : bend;
      d += across ? ` q ${step / 2} ${out} ${step} 0` : ` q ${out} ${step / 2} 0 ${step}`;
    }
    return d;
  }

  /** Width and color of each side (top, right, bottom, left) of a square. */
  private sides(shape: GridShape, square: number): Array<[number, string]> {
    const { size, sectionOf } = shape;
    const row = Math.floor(square / size);
    const col = square % size;
    const mine = sectionOf[square];
    const at = (r: number, c: number) => (r < 0 || c < 0 || r >= size || c >= size ? -1 : sectionOf[r * size + c]);
    return [at(row - 1, col), at(row, col + 1), at(row + 1, col), at(row, col - 1)].map((next) =>
      next === mine ? [THIN, 'var(--grid-line)'] : next < 0 ? [EDGE, 'var(--navy)'] : [HALF, 'var(--navy)'],
    );
  }
}
```

```ts
/** Word Grid: Knotwords in big print, with a tray of letters on top to tap or drag into place. */
export class GridScreen implements GameScreen {
  readonly id = 'grid';
  readonly title = 'Word Grid';
  /** True while the words are turning green or she's celebrating. */
  busy = false;
  private host!: FrameHost;
  private saved!: SavedGrid;
  private game!: GridGame;
  private selected = 0;
  private held: Held | null = null;
  private view: GridView | null = null;
  private layout: GridLayout | null = null;
  private idleTimer = 0;
  private readonly isWord: (word: string) => boolean;

  constructor(

  ) {

  }

  start(host: FrameHost): void {

  }

  render(layer: HTMLElement, frame: FrameLayout, animateIn: boolean): GameTargets {

    this.view?.setPressed(id, false);
  }

  /**
   * She picked up a letter. Its targets wait until she lets go: telling the touch tracker about new targets now
   * would cancel the drag.
   */
  dragStarted(id: string): void {
    if (this.busy || !this.view) return;
    if (id.startsWith('tray:')) {
      const index = Number(id.slice('tray:'.length));
      const tile = this.trayLetters()[index];
      if (!tile || tile.used) return;
      this.held = { from: 'tray', index, letter: tile.letter };
    } else if (id.startsWith('square:')) {
      const square = Number(id.slice('square:'.length));
      const letter = this.game.letterAt(square);
      if (!letter || this.game.isHinted(square)) return;
      // Picking up a letter selects its square, so the tray shows its section.
      this.selected = square;
      this.view.update(this.viewState());
      this.held = { from: 'square', square, letter };
    } else {
      return;
    }
    this.view.startDrag(this.held.letter, id);
  }

  dragMoved(_id: string, x: number, y: number): void {
    if (this.held) this.view?.moveDrag(x, y);
  }

  dragEnded(_id: string, x: number, y: number): void {
    const held = this.held;
    const layout = this.layout;
    this.held = null;
    if (!held || !layout || !this.view) return;
    const target = this.game.shape.squares.find((square) => contains(layout.squares[square], x, y)) ?? -1;
    const home = this.game.sectionOf(held.from === 'tray' ? this.selected : held.square);
    const inSection = target >= 0 && this.game.sectionOf(target) === home;
    if (held.from === 'tray') {
      if (inSection && this.game.place(target, held.letter)) return this.landed(target);
      // Anywhere else: back to its place in the tray.
      this.view.dropDrag(trayTiles(layout, this.trayLetters().length)[held.index]);
      return this.refresh();
    }
    if (inSection && target !== held.square && this.game.move(held.square, target)) return this.landed(target);
    if (inSection) {
      // Its own square, or a hinted one: it stays where it was.
      this.view.dropDrag(layout.squares[held.square]);
      return this.refresh();
    }
    // Off the grid, or onto another section: back to the tray.
    this.game.remove(held.square);
    this.view.dropDrag(this.trayRectFor(held.letter));
    this.changed();
  }

  /** The iPad cancelled the touch (or the screen changed): put the letter back. Targets are unchanged. */
  dragCancelled(): void {
    this.held = null;
    this.view?.dropDrag(null);
    this.view?.update(this.viewState());
  }

  pause(): void {
    window.clearTimeout(this.idleTimer);
    this.view?.setHintPulse(false);
    if (this.held) this.dragCancelled();
    this.save();
  }

  resume(): void {
    if (!this.busy) this.resetIdle();
  }

  private loadPuzzle(): void {
    const level = this.saved.level;
    const progress = progressFor(this.saved, this.puzzles, level);
    this.game = new GridGame(this.puzzles[level][progress.index], progress.placed, progress.hinted, this.isWord);
    this.saved = { ...this.saved, levels: { ...this.saved.levels, [level]: progress } };
    this.selected = this.game.firstEmpty();
  }

  private save(): void {
    const level = this.saved.level;
    const progress = progressFor(this.saved, this.puzzles, level);
    const current = { ...progress, placed: this.game.placed, hinted: this.game.hinted };
    this.saved = { ...this.saved, levels: { ...this.saved.levels, [level]: current } };
    saveSavedGrid(this.store, this.saved);
  }

  private trayLetters(): TrayLetter[] {
    return this.game.tray(this.game.sectionOf(this.selected));
  }

  /** Where a letter goes back to in the tray: a free tile with that letter. */
  private trayRectFor(letter: string): Rect | null {
    if (!this.layout) return null;
    const tray = this.trayLetters();
    const index = tray.findIndex((tile) => !tile.used && tile.letter === letter);
    return trayTiles(this.layout, tray.length)[Math.max(0, index)] ?? null;
  }

  private viewState(): GridViewState {
    const { shape } = this.game;
    return {
      level: this.saved.level,
      shape,
      letters: shape.open.map((_, square) => this.game.letterAt(square)),
      hinted: new Set(this.game.hinted),
      selected: this.selected,
      tray: this.trayLetters(),
      corners: shape.sections.map((_, section) => this.game.tray(section)),
      badRuns: this.game.badRuns().map((i) => shape.runs[i]),
    };
  }

  /** Placed letters can be picked up; empty and hinted squares can only be tapped (to select them). */
  private targets(): GameTargets {
    const layout = this.layout;
    if (!layout) return { tiles: [], buttons: [] };
    const buttons: ButtonGeometry[] = [
      ...GRID_LEVELS.map((level, i) => ({ id: `level:${level}`, ...layout.levels[i] })),
      { id: 'hint', ...layout.hintButton },
    ];
    const draggables: ButtonGeometry[] = [];
    for (const square of this.game.shape.squares) {
      const target = { id: `square:${square}`, ...layout.squares[square] };
      if (this.game.letterAt(square) && !this.game.isHinted(square)) draggables.push(target);
      else buttons.push(target);
    }
    const tray = this.trayLetters();
    trayTiles(layout, tray.length).forEach((rect, i) => {
      if (!tray[i].used) draggables.push({ id: `tray:${i}`, ...rect });
    });
    return { tiles: [], buttons, draggables };
  }

  /** Redraws the squares and tray and tells the touch tracker what can be touched now. */
  private refresh(): void {
    this.view?.update(this.viewState());
    this.host.updateTargets(this.targets());
  }

  /** A letter landed in `square` from a drag. */
  private landed(square: number): void {
    this.selected = square;
    this.view?.dropDrag(null);
    this.changed();
    this.view?.popSquare(square);
  }

  /** Something on the grid changed: save it, show it, and see if she's finished. */
  private changed(): void {
    this.save();
    this.resetIdle();
    this.refresh();
    if (this.game.solved) void this.finish();
  }

  private select(square: number): void {
    if (!this.game.isOpen(square)) return;
    this.selected = square;
    this.refresh();
  }

  private tapTray(index: number): void {
    const tile = this.trayLetters()[index];
    const square = this.selected;
    if (!tile || tile.used || !this.game.place(square, tile.letter)) return;
    this.selected = this.game.nextEmptyInSection(square) ?? square;
    this.changed();
    this.view?.popSquare(square);
  }

  private hint(): void {
    const square = this.game.hint(this.selected);
    if (square === null) return;
    this.selected = square;
    this.changed();
    this.view?.popSquare(square);
  }

  /** Each level keeps its own puzzle and letters, so switching never loses anything. */
  private chooseLevel(level: GridLevel): void {
    if (level === this.saved.level || !GRID_LEVELS.includes(level)) return;
    this.save();
    this.saved = { ...this.saved, level };
    this.loadPuzzle();
    this.save();
    this.host.redraw(true);
    this.resetIdle();
  }

  private async finish(): Promise<void> {
    this.busy = true;
    window.clearTimeout(this.idleTimer);
    this.view?.setHintPulse(false);
    await this.view?.finish(this.game.shape.runs);
    const hinted = this.game.hinted.length > 0;
    this.saved = { ...this.saved, stats: recordGridSolve(this.saved.stats, dayKey(this.now())) };
    const streakDays = this.host.recordPlayDay();
    // Save the next puzzle as her place now, so closing the app mid-celebration loses nothing.
    this.saved = nextPuzzle(this.saved, this.puzzles, this.saved.level);
    this.loadPuzzle();
    this.save();
    await this.host.celebrate({
      lines: gridLines({
        hinted,
        streakDays,
        todaySolved: this.saved.stats.todaySolved,
        puzzleNumber: this.saved.stats.totalSolved,
      }),
      nextLabel: 'Next puzzle',
      timeoutMs: CELEBRATION_MS,
    });
    this.busy = false;
    this.host.redraw(true);
    this.resetIdle();
  }

  private resetIdle(): void {
    window.clearTimeout(this.idleTimer);
    this.view?.setHintPulse(false);
    this.idleTimer = window.setTimeout(() => this.view?.setHintPulse(true), IDLE_HINT_MS);
  }
}
```
