> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Jigsaw, a fourth game: big-piece jigsaws of the starter pictures, played in Tray, Board or Table style with 9, 16 or 30 pieces, with a "Pick a picture" page, saving, Hint, stats and a starter-picture pipeline.

**Architecture:** A `GameScreen` in `src/jigsaw/`, built like Word Grid: pure modules (piece shapes, layout, rules, saving, picture list, good news) with Vitest tests, plus a view, a picture page and a controller that are checked in the browser. Pieces are cut once from the photo into canvases and moved with absolute positioning. Starter pictures are made by a Node script (`sharp`) from originals kept out of git, published in `public/jigsaw/` and precached by the service worker.

**Tech Stack:** TypeScript 5.6, Vite 5, Vitest 2, vite-plugin-pwa, `sharp` 0.33.5 (new dev dependency, scripts only).

**Spec:** `docs/superpowers/specs/2026-10-28-jigsaw-design.md`

## Read first

- **Two plans.** The spec's build order is split. **This plan** is spec steps 1, 2 and 5 plus Jigsaw appearing for every family: the whole game with the starter pictures, shippable on its own. **Part 2** (spec steps 3–4: family photos from the iPad's library, "Your new puzzle", the settings page's "Jigsaw pictures" section) gets its own plan once this one is live, so it builds on real code. This plan already leaves room for it: `picturePage` takes a page size so the Add a photo tile can take a place, and picture ids are prefixed (`starter:`), ready for `family:`.
- **Small changes from the spec, made while planning:**
  1. **Table style has no tap-then-tap**; tapping a table piece gives it a little wiggle (it shows it can be moved). In **Board** style, tapping a lifted piece's wrong spot on the board leaves it lying on that spot; tapping outside the board does nothing. The touch tracker reports which button a tap hit, not where, so "tap anywhere to put it there" isn't possible without changing the tracker for every game.
  2. A lifted piece uses Word Grid's orange held look (`--select`), not purple, so "held" looks the same in both games.
  3. The finished photo grows to fill the space above a full-width caption strip; the "Well done!" card's button says **Next picture** and opens the picture page.
  4. Table pieces start in a neat grid of places (each piece in a shuffled one), not a jittered grid. Nothing touches, which is what mattered.
  5. Hint pulses after two minutes without a move, as in the other games.

```ts
import { seededRandom } from '../shared/random';

/** One side of a piece: flat (the edge of the picture), a tab sticking out, or a hole cut in. */
export type Edge = 0 | 1 | -1;

/** A piece's place in the picture and its four sides. Piece i is at column i % cols, row floor(i / cols). */
export interface PieceShape {
  col: number;
  row: number;
  top: Edge;
  right: Edge;
  bottom: Edge;
  left: Edge;
}

/** How far a tab reaches, as a share of the piece's shorter side. */
export const TAB_SHARE = 0.2;

export function tabFor(w: number, h: number): number {
  return TAB_SHARE * Math.min(w, h);
}

/** A number from the picture and its piece count, so the same puzzle always cuts (and deals) the same way. */
export function pieceSeed(pictureId: string, count: number): number {
  const text = `${pictureId}:${count}`;
  let hash = 0x811c9dc5;
  for (let i = 0; i < text.length; i++) {
    hash ^= text.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193);
  }
  return hash >>> 0;
}

const flip = (edge: Edge): Edge => (edge === 1 ? -1 : edge === -1 ? 1 : 0);

/** Every inside edge gets a tab on one side and the matching hole on the other; the picture's edges stay flat. */
export function cutPieces(cols: number, rows: number, seed: number): PieceShape[] {
  const random = seededRandom(seed);
  const pieces: PieceShape[] = Array.from({ length: cols * rows }, (_, i) => ({
    col: i % cols,
    row: Math.floor(i / cols),
    top: 0,
    right: 0,
    bottom: 0,
    left: 0,
  }));
  pieces.forEach((piece, i) => {
    if (piece.col < cols - 1) {
      piece.right = random() < 0.5 ? 1 : -1;
      pieces[i + 1].left = flip(piece.right);
    }
    if (piece.row < rows - 1) {
      piece.bottom = random() < 0.5 ? 1 : -1;
      pieces[i + cols].top = flip(piece.bottom);
    }
  });
  return pieces;
}

const n = (value: number) => String(Math.round(value * 100) / 100);

/** One side from (x0, y0) to (x1, y1); (nx, ny) points out of the piece. A tab is a round knob on a narrower neck. */
function side(x0: number, y0: number, x1: number, y1: number, nx: number, ny: number, edge: Edge, tab: number): string {
  if (edge === 0) return ` L ${n(x1)} ${n(y1)}`;
  const at = (along: number, out: number) =>
    `${n(x0 + (x1 - x0) * along + nx * out * tab * edge)} ${n(y0 + (y1 - y0) * along + ny * out * tab * edge)}`;
  return (
    ` L ${at(0.35, 0)}` +
    ` C ${at(0.4, 0)} ${at(0.42, 0.35)} ${at(0.37, 0.55)}` +
    ` C ${at(0.32, 0.8)} ${at(0.4, 1)} ${at(0.5, 1)}` +
    ` C ${at(0.6, 1)} ${at(0.68, 0.8)} ${at(0.63, 0.55)}` +
    ` C ${at(0.58, 0.35)} ${at(0.6, 0)} ${at(0.65, 0)}` +
    ` L ${at(1, 0)}`
  );
}

/**
 * A piece's outline as an SVG path, clockwise from its top-left corner. The body (w × h) sits at (tab, tab), so the
 * whole piece, tabs included, fits in (w + 2·tab) × (h + 2·tab).
 */
export function piecePath(shape: PieceShape, w: number, h: number, tab: number): string {
  const [l, t, r, b] = [tab, tab, tab + w, tab + h];
  return (
    `M ${n(l)} ${n(t)}` +
    side(l, t, r, t, 0, -1, shape.top, tab) +
    side(r, t, r, b, 1, 0, shape.right, tab) +
    side(r, b, l, b, 0, 1, shape.bottom, tab) +
    side(l, b, l, t, -1, 0, shape.left, tab) +
    ' Z'
  );
}
```

```ts
import type { Rect } from '../frame/layout';

/** Where the frame sits in the photo's spare room: 0 is the left or top, 1 the right or bottom. */
export interface CropAt {
  x: number;
  y: number;
}

/** A frame starts centered across and a quarter of the way down, where faces usually are. */
export const START_CROP: CropAt = { x: 0.5, y: 0.25 };

const share = (value: number) => Math.min(1, Math.max(0, value));

/** The largest 4:3 rectangle inside a width × height photo, placed `at` within the spare room. */
export function cropRect(width: number, height: number, at: CropAt = START_CROP): Rect {
  // 4:3 in whole-number steps: 1500 * (4 / 3) in floating point can land a hair under 2000.
  const w = Math.floor(Math.min(width, (height * 4) / 3));
  const h = Math.floor((w * 3) / 4);
  return { x: Math.round((width - w) * share(at.x)), y: Math.round((height - h) * share(at.y)), w, h };
}
```

```ts
import { START_CROP, type CropAt } from '../src/jigsaw/crop';

/** What captions.txt says about one photo. */
export interface StarterNote {
  caption: string;
  keep: CropAt;
}

/** Which part of a photo to keep, from words like "top", "bottom left" or "middle". */
export function keepAt(words: string): CropAt {
  const keep = { ...START_CROP };
  for (const word of words.toLowerCase().split(/[\s,]+/)) {
    if (word === 'left') keep.x = 0;
    else if (word === 'center') keep.x = 0.5;
    else if (word === 'right') keep.x = 1;
    else if (word === 'top') keep.y = 0;
    else if (word === 'middle') keep.y = 0.5;
    else if (word === 'bottom') keep.y = 1;
  }
  return keep;
}

/** captions.txt: one line per photo, "file name | caption | keep". Blank lines and lines starting with # are skipped. */
export function parseCaptions(text: string): Map<string, StarterNote> {
  const notes = new Map<string, StarterNote>();
  for (const line of text.split(/\r?\n/)) {
    const trimmed = line.trim();
    if (!trimmed || trimmed.startsWith('#')) continue;
    const [file, caption = '', keep = ''] = trimmed.split('|').map((part) => part.trim());
    if (file) notes.set(file, { caption, keep: keepAt(keep) });
  }
  return notes;
}

/** A picture's id from its file name: "Lake Louise.JPG" is "lake-louise". */
export function slug(file: string): string {
  const name = file
    .replace(/\.[^.]+$/, '')
    .toLowerCase()
    .replace(/['’]/g, '')
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '');
  return name || 'picture';
}

/** Ids for the files, in order; a name used twice gets "-2", "-3"… */
export function uniqueIds(files: readonly string[]): string[] {
  const used = new Map<string, number>();
  return files.map((file) => {
    const base = slug(file);
    const count = (used.get(base) ?? 0) + 1;
    used.set(base, count);
    return count === 1 ? base : `${base}-${count}`;
  });
}

/** The photos in captions.txt come first, in its order; the others follow by name. */
export function starterOrder(files: readonly string[], listed: readonly string[]): string[] {
  const first = listed.filter((file) => files.includes(file));
  const rest = files.filter((file) => !first.includes(file)).sort((a, b) => a.localeCompare(b));
  return [...first, ...rest];
}
```

```ts
/** iPhone photos (HEIC) become JPEGs first, with macOS's own sips. */
function readable(file: string, scratch: string): string {
  const path = join(SOURCE, file);
  if (!/\.hei[cf]$/i.test(file)) return path;
  const out = join(scratch, `${file}.jpg`);
  execFileSync('sips', ['-s', 'format', 'jpeg', path, '--out', out], { stdio: 'ignore' });
  return out;
}

const escape = (text: string) => text.replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' })[c]!);

    for (const [i, file] of files.entries()) {
      const id = ids[i];
      // Turned the right way up first: phones save sideways photos with a note saying which way to turn them.
      const { data, info } = await sharp(readable(file, scratch)).rotate().toBuffer({ resolveWithObject: true });
      const note = notes.get(file);

      const framed = sharp(data).extract({ left: crop.x, top: crop.y, width: crop.w, height: crop.h });
      await framed.clone().resize(PHOTO.w, PHOTO.h).jpeg({ quality: 82, mozjpeg: true }).toFile(join(OUT, `${id}.jpg`));
      await framed.clone().resize(THUMB.w, THUMB.h).jpeg({ quality: 80, mozjpeg: true }).toFile(join(OUT, `${id}-thumb.jpg`));
      starters.push({ id, file: `${id}.jpg`, thumb: `${id}-thumb.jpg`, caption: note?.caption ?? '' });
      console.log(`${file} → ${id}`);

    rmSync(scratch, { recursive: true, force: true });
```

```ts
import type { Picture } from './types';

/** "Pick a picture" shows six places: three across, two down. */
export const PAGE_SIZE = 6;

const ID = /^[a-z0-9-]+$/;

/** The published starters.json, checked entry by entry; anything odd is left out. `base` is the site's address. */
export function parseStarters(value: unknown, base: string): Picture[] {
  if (!Array.isArray(value)) return [];
  const pictures: Picture[] = [];
  const seen = new Set<string>();
  for (const entry of value) {
    if (typeof entry !== 'object' || entry === null) continue;
    const { id, file, thumb, caption } = entry as Record<string, unknown>;
    if (typeof id !== 'string' || !ID.test(id) || seen.has(id) || typeof file !== 'string' || typeof thumb !== 'string') continue;
    seen.add(id);
    pictures.push({
      id: `starter:${id}`,
      src: `${base}jigsaw/${file}`,
      thumb: `${base}jigsaw/${thumb}`,
      caption: typeof caption === 'string' ? caption.trim() : '',
    });
  }
  return pictures;
}

/** The starter pictures; none if the list can't be had (offline before the app ever saved it). */
export async function loadStarters(base: string, fetcher: (url: string) => Promise<Response> = (url) => fetch(url)): Promise<Picture[]> {
  try {
    const response = await fetcher(`${base}jigsaw/starters.json`);
    return response.ok ? parseStarters(await response.json(), base) : [];
  } catch {
    return [];
  }
}

export interface PicturePage {
  pictures: Picture[];
  /** The first one's place in the whole list. */
  first: number;
  hasBack: boolean;
  hasMore: boolean;
}

/** One page of `perPage` pictures (fewer than six when the page also holds the Add a photo tile). */
export function picturePage(pictures: readonly Picture[], page: number, perPage = PAGE_SIZE): PicturePage {
  const pages = Math.max(1, Math.ceil(pictures.length / perPage));
  const p = Math.min(Math.max(0, page), pages - 1);
  const first = p * perPage;
  return { pictures: pictures.slice(first, first + perPage), first, hasBack: p > 0, hasMore: p < pages - 1 };
}
```

```ts
import { reviewButtonIn, type FrameLayout, type Rect } from '../frame/layout';
import { tabFor } from './cut';
import { GRID_SIZES, PICTURE_ASPECT, type PieceCount } from './types';

/** Space between Hint, Pictures, the tray, the board, the rows and the top-right buttons. */
const GAP = 16;
const SLOT_COUNT = 3;
const SLOT_GAP = 12;
const SLOT_PADDING = 6;
/** Tray, Board and Table sit this far above 9, 16 and 30. */
const ROW_GAP = 8;
/** Table pieces start at least this far apart. */
const TABLE_GAP = 8;
/** Table pieces are never smaller than this (the free area always has room well above it). */
const TABLE_MIN = 40;
/** The finished picture's caption: up to two lines across the bottom. */
const CAPTION_HEIGHT = 96;
/** "Pick a picture". */
const TITLE_HEIGHT = 72;
const TILE_GAP = 24;
const TILE_CAPTION = 34;
const PAGE_BUTTON_HEIGHT = 96;
const BACK_WIDTH = 200;
const CLOSE_WIDTH = 180;
const MORE_WIDTH = 260;

export interface JigsawLayout {
  frame: FrameLayout;
  hintButton: Rect;
  /** The game's own purple button, beside Hint. */
  picturesButton: Rect;
  /** The yellow backing: the tray (Tray and Board) or the box lid's place (Table). */
  tray: Rect;
  /** The tray's three square places for pieces. */
  slots: Rect[];
  /** The whole picture, small, in the tray's place (Table). */
  lid: Rect;
  /** Tray, Board, Table: just above the piece counts. */
  styleButtons: Rect[];
  /** 9, 16, 30: the frame's level row. */
  countButtons: Rect[];
  /** The 4:3 board (Tray and Board). */
  board: Rect;
  /** Where loose pieces may lie: everything clear of the buttons, the tray and the rows. The table starts at free[0]. */
  free: Rect[];
  /** Where the finished picture is shown, and its caption under it. */
  finished: Rect;
  caption: Rect;
  /** Next, over the tray's place, while she looks back at a finished picture. */
  reviewButton: Rect;
}

/** The largest rectangle of `aspect` (width / height) in `area`, centered in it. */
function fit(area: Rect, aspect: number): Rect {
  const w = Math.min(area.w, area.h * aspect);
  const h = w / aspect;
  return { x: area.x + (area.w - w) / 2, y: area.y + (area.h - h) / 2, w, h };
}

export function computeJigsawLayout(frame: FrameLayout): JigsawLayout {
  // Hint top left and the piece counts in the level row, as in every game; Pictures beside Hint (where Word Circle
  // has Shuffle); the tray between Pictures and the top-right buttons; Tray, Board and Table above the counts.
  const { play, hintButton, levelButtons: countButtons, corner } = frame;
  const picturesButton = { ...hintButton, x: hintButton.x + hintButton.w + GAP };
  const trayLeft = picturesButton.x + picturesButton.w + GAP;
  const tray = { x: trayLeft, y: play.y, w: corner.x - GAP - trayLeft, h: hintButton.h };
  const slot = Math.floor(Math.min(tray.h - 2 * SLOT_PADDING, (tray.w - (SLOT_COUNT + 1) * SLOT_GAP) / SLOT_COUNT));
  const slotsLeft = tray.x + (tray.w - (SLOT_COUNT * slot + (SLOT_COUNT - 1) * SLOT_GAP)) / 2;
  const slots = Array.from({ length: SLOT_COUNT }, (_, i) => ({ x: slotsLeft + i * (slot + SLOT_GAP), y: tray.y + (tray.h - slot) / 2, w: slot, h: slot }));
  const lid = fit({ x: tray.x, y: tray.y + SLOT_PADDING, w: tray.w, h: tray.h - 2 * SLOT_PADDING }, PICTURE_ASPECT);
  const styleButtons = countButtons.map((rect) => ({ ...rect, y: rect.y - ROW_GAP - rect.h }));

  const top = hintButton.y + hintButton.h + GAP;
  const rowsTop = styleButtons[0].y;
  const main = { x: play.x, y: top, w: play.w, h: rowsTop - GAP - top };
  const last = countButtons[countButtons.length - 1];
  const besideRowsLeft = last.x + last.w + GAP;
  const besideRows = { x: besideRowsLeft, y: rowsTop, w: play.x + play.w - besideRowsLeft, h: play.y + play.h - rowsTop };
  const caption = { x: play.x, y: play.y + play.h - CAPTION_HEIGHT, w: play.w, h: CAPTION_HEIGHT };
  return {
    frame,
    hintButton,
    picturesButton,
    tray,
    slots,
    lid,
    styleButtons,
    countButtons,
    board: fit(main, PICTURE_ASPECT),
    free: [main, besideRows],
    finished: fit({ x: play.x, y: top, w: play.w, h: caption.y - GAP - top }, PICTURE_ASPECT),
    caption,
    reviewButton: reviewButtonIn(tray),
  };
}

/** Table pieces' size, and a starting place (a body rect) for each, apart, in the free area. */
export interface TablePieces {
  w: number;
  h: number;
  spots: Rect[];
}

/** Cells of cellW × cellH across `area`, row by row, the leftover space shared out around them. */
function cells(area: Rect, cellW: number, cellH: number): Rect[] {
  const across = Math.floor(area.w / cellW);
  const down = Math.floor(area.h / cellH);
  const x0 = area.x + (area.w - across * cellW) / 2;
  const y0 = area.y + (area.h - down * cellH) / 2;
  return Array.from({ length: across * down }, (_, i) => ({ x: x0 + (i % across) * cellW, y: y0 + Math.floor(i / across) * cellH, w: cellW, h: cellH }));
}

/** The biggest table pieces that still leave room for every piece, apart, in the free area (never bigger than on the board). */
export function tablePieces(layout: JigsawLayout, count: PieceCount): TablePieces {
  const { cols, rows } = GRID_SIZES[count];
  const shape = layout.board.h / rows / (layout.board.w / cols);
  let w = Math.floor(layout.board.w / cols);
  for (;;) {
    const h = w * shape;
    const tab = tabFor(w, h);
    const cellW = w + 2 * tab + TABLE_GAP;
    const cellH = h + 2 * tab + TABLE_GAP;
    const room = layout.free.flatMap((area) => cells(area, cellW, cellH));
    if (room.length >= count || w <= TABLE_MIN) {
      return { w, h, spots: room.map((cell) => ({ x: cell.x + (cellW - w) / 2, y: cell.y + (cellH - h) / 2, w, h })) };
    }
    w -= 1;
  }
}

const distance = (area: Rect, x: number, y: number) =>
  Math.hypot(Math.max(area.x - x, 0, x - (area.x + area.w)), Math.max(area.y - y, 0, y - (area.y + area.h)));

/** Moves a piece's body (or a group's bounds) inside the part of the free area its middle is nearest. */
export function clampToFree(free: readonly Rect[], body: Rect): Rect {
  const cx = body.x + body.w / 2;
  const cy = body.y + body.h / 2;
  const area = free.reduce((best, a) => (distance(a, cx, cy) < distance(best, cx, cy) ? a : best));
  return {
    ...body,
    x: Math.max(area.x, Math.min(body.x, area.x + area.w - body.w)),
    y: Math.max(area.y, Math.min(body.y, area.y + area.h - body.h)),
  };
}

export function computePicturePageLayout(frame: FrameLayout): PicturePageLayout {
  const { play, corner } = frame;
```

```ts
import { seededRandom, shuffle } from '../shared/random';
import { GRID_SIZES, type JigsawProgress, type JigsawStyle, type PieceCount, type Point } from './types';

/** The tray holds this many pieces at a time. */
export const TRAY_SIZE = 3;
/** A piece dropped this close to its spot (in pieces, across and down) clicks in. */
export const SNAP = 0.4;
/** On the table, a piece this close to where it belongs beside another joins it. */
export const JOIN = 0.3;

/** What a drop did: clicked in, went back to the tray, was left lying, joined others on the table, or just moved. */
export type DropResult = 'placed' | 'back' | 'loose' | 'joined' | 'moved';

/** The order the tray hands pieces out: the corners, then the other edge pieces, then the middle, each shuffled. */
export function trayOrder(cols: number, rows: number, random: () => number): number[] {
  const corners: number[] = [];
  const edges: number[] = [];
  const middle: number[] = [];
  for (let id = 0; id < cols * rows; id++) {
    const col = id % cols;
    const row = Math.floor(id / cols);
    const sides = Number(col === 0 || col === cols - 1) + Number(row === 0 || row === rows - 1);
    (sides === 2 ? corners : sides === 1 ? edges : middle).push(id);
  }
  return [...shuffle(corners, random), ...shuffle(edges, random), ...shuffle(middle, random)];
}

const range = (n: number) => Array.from({ length: n }, (_, i) => i);
const near = (a: Point, b: Point, limit: number) => Math.abs(a.x - b.x) <= limit && Math.abs(a.y - b.y) <= limit;
const isPermutation = (ids: readonly number[], count: number) =>
  ids.length === count && new Set(ids).size === count && ids.every((id) => Number.isInteger(id) && id >= 0 && id < count);

/** True if a saved place belongs to this style and piece count. */
function fits(style: JigsawStyle, count: PieceCount, p: JigsawProgress): boolean {
  const { cols } = GRID_SIZES[count];
  if (style === 'table') {
    if (p.positions.length !== count || p.groups.length !== count || !isPermutation(p.order, count)) return false;
    // Joined pieces must still lie exactly side by side.
    return p.groups.every((group, i) => {
      const first = p.groups.indexOf(group);
      const dx = p.positions[i][0] - p.positions[first][0] - ((i % cols) - (first % cols));
      const dy = p.positions[i][1] - p.positions[first][1] - (Math.floor(i / cols) - Math.floor(first / cols));
      return Math.abs(dx) < 0.01 && Math.abs(dy) < 0.01;
    });
  }
  const inTray = p.tray.filter((id): id is number => id !== null);
  const every = [...p.placed, ...p.queue, ...inTray, ...p.loose.map(([id]) => id)];
  return p.tray.length === TRAY_SIZE && isPermutation(every, count) && (style === 'board' || p.loose.length === 0);
}

/**
 * One puzzle being played. Positions are in pieces (x in piece widths, y in piece heights) for a piece's body, from
 * the board's top-left corner (Tray and Board) or the table's (Table). A piece's own spot is (col, row).
 */
export class JigsawGame {
  readonly cols: number;
  readonly rows: number;
  hints: number;
  private readonly placedSet: Set<number>;
  private readonly queue: number[];
  private readonly slots: Array<number | null>;
  /** Pieces lying around, bottom first (a Map keeps the order they were put down). */
  private readonly looseMap: Map<number, Point>;
  private readonly positions: Point[];
  private readonly groups: number[];
  private order: number[];

  private constructor(
    readonly style: JigsawStyle,
    readonly count: PieceCount,
    progress: JigsawProgress,
  ) {
    const { cols, rows } = GRID_SIZES[count];
    this.cols = cols;
    this.rows = rows;
    this.hints = progress.hints;
    this.placedSet = new Set(progress.placed);
    this.queue = [...progress.queue];
    this.slots = [...progress.tray];
    this.looseMap = new Map(progress.loose.map(([id, x, y]) => [id, { x, y }]));
    this.positions = progress.positions.map(([x, y]) => ({ x, y }));
    this.groups = [...progress.groups];
    this.order = [...progress.order];
  }

  /** A new puzzle, dealt from `seed`. Table pieces start in `spots` (one each), shuffled. */
  static fresh(style: JigsawStyle, count: PieceCount, seed: number, spots: readonly Point[] = []): JigsawGame {
    const { cols, rows } = GRID_SIZES[count];
    // Not the same numbers as the cut, so which pieces come first isn't tied to their shapes.
    const random = seededRandom(seed ^ 0x5bd1e995);
    const empty: JigsawProgress = { placed: [], queue: [], tray: [], loose: [], positions: [], groups: [], order: [], hints: 0, elapsedMs: 0 };
    if (style === 'table') {
      const where = shuffle(spots.length > 0 ? spots : [{ x: 0, y: 0 }], random);
      const positions = range(count).map((id): [number, number] => [where[id % where.length].x, where[id % where.length].y]);
      return new JigsawGame(style, count, { ...empty, positions, groups: range(count), order: range(count) });
    }
    const order = trayOrder(cols, rows, random);
    return new JigsawGame(style, count, { ...empty, queue: order.slice(TRAY_SIZE), tray: order.slice(0, TRAY_SIZE) });
  }

  /** Her saved place, or null if it doesn't fit this puzzle (start afresh). */
  static restore(style: JigsawStyle, count: PieceCount, progress: JigsawProgress): JigsawGame | null {
    return fits(style, count, progress) ? new JigsawGame(style, count, progress) : null;
  }

  /** A piece's own spot. */
  home(id: number): Point {
    return { x: id % this.cols, y: Math.floor(id / this.cols) };
  }

  isPlaced(id: number): boolean {
    return this.placedSet.has(id);
  }

  /** Pieces in their spots, in the order they went in. */
  get placed(): number[] {
    return [...this.placedSet];
  }

  /** The tray's places, left to right: a piece, or null. */
  get tray(): ReadonlyArray<number | null> {
    return this.slots;
  }

  /** Pieces lying around (Board), bottom first. */
  get loose(): Array<[number, Point]> {
    return [...this.looseMap].map(([id, p]) => [id, { ...p }]);
  }

  /** Where a table piece is. */
  positionOf(id: number): Point {
    return { ...this.positions[id] };
  }

  /** Table pieces, bottom first. */
  get stacking(): readonly number[] {
    return this.order;
  }

  /** The pieces joined to `id`, itself included, bottom first (just itself off the table). */
  groupOf(id: number): number[] {
    if (this.style !== 'table') return [id];
    return this.order.filter((p) => this.groups[p] === this.groups[id]);
  }

  get solved(): boolean {
    if (this.style === 'table') return this.groups.every((group) => group === this.groups[0]);
    return this.placedSet.size === this.count;
  }

  /** She let go of piece `id` with its body at `at`. */
  drop(id: number, at: Point): DropResult {
    if (this.style === 'table') return this.dropOnTable(id, at);
    if (this.placedSet.has(id)) return 'back';
    if (near(at, this.home(id), SNAP)) {
      this.place(id);
      return 'placed';
    }
    if (this.style === 'tray') return 'back';
    this.takeOut(id);
    this.looseMap.set(id, { ...at });
    return 'loose';
  }

  /** Hint: places the tray's first piece (or, with the tray empty, the bottom one lying around). Returns it, or null when done. */
  hint(): number | null {
    if (this.solved) return null;
    if (this.style === 'table') return this.hintOnTable();
    const id = this.slots.find((slot) => slot !== null) ?? this.looseMap.keys().next().value;
    if (id === undefined) return null;
    this.place(id);
    this.hints++;
    return id;
  }

  progress(elapsedMs: number): JigsawProgress {
    return {
      placed: [...this.placedSet],
      queue: [...this.queue],
      tray: [...this.slots],
      loose: [...this.looseMap].map(([id, p]) => [id, p.x, p.y]),
      positions: this.positions.map((p) => [p.x, p.y]),
      groups: [...this.groups],
      order: [...this.order],
      hints: this.hints,
      elapsedMs: Math.round(elapsedMs),
    };
  }

  private place(id: number): void {
    this.takeOut(id);
    this.placedSet.add(id);
  }

  /** Takes a piece off the tray (the next one slides into its place) or up from where it was lying. */
  private takeOut(id: number): void {
    const slot = this.slots.indexOf(id);
    if (slot >= 0) this.slots[slot] = this.queue.shift() ?? null;
    this.looseMap.delete(id);
  }

  // Table style: Task 7.
  private dropOnTable(_id: number, _at: Point): DropResult {
    return 'moved';
  }

  private hintOnTable(): number | null {
    return null;
  }
}
```

```ts
    };
  }

  /** The pieces above, right, below and left of `p` in the picture. */
  private neighbors(p: number): number[] {
    const col = p % this.cols;
    const row = Math.floor(p / this.cols);
    const out: number[] = [];
    if (row > 0) out.push(p - this.cols);
    if (col < this.cols - 1) out.push(p + 1);
    if (row < this.rows - 1) out.push(p + this.cols);
    if (col > 0) out.push(p - 1);
    return out;
  }

  /** A piece of `members` and a neighbor outside the group that lies close to where it belongs. */
  private closePair(members: readonly number[]): [number, number] | null {
    for (const p of members) {
      for (const q of this.neighbors(p)) {
        if (this.groups[q] !== this.groups[p] && near(this.positions[q], this.beside(p, q), JOIN)) return [p, q];
      }
    }
    return null;
  }

  /** The first join moves what she dropped onto the piece it fits; anything else that then fits is pulled in. */
  private joinAll(dropped: number[]): boolean {
    let members = dropped;
    let joined = false;
    for (let pair = this.closePair(members); pair; pair = this.closePair(members)) {
      const [p, q] = pair;
      const target = this.beside(p, q);
      const other = this.groupOf(q);
      if (joined) this.shift(other, target.x - this.positions[q].x, target.y - this.positions[q].y);
      else this.shift(members, this.positions[q].x - target.x, this.positions[q].y - target.y);
      members = this.merge(members, other);
      joined = true;
    }
    if (joined) this.raise(members);
    return joined;
  }

  /** One group from two; it takes the lower group number. */
  private merge(a: readonly number[], b: readonly number[]): number[] {
    const group = Math.min(this.groups[a[0]], this.groups[b[0]]);
    for (const p of [...a, ...b]) this.groups[p] = group;
    return this.groupOf(a[0]);
  }

  private shift(ids: readonly number[], dx: number, dy: number): void {
    for (const p of ids) this.positions[p] = { x: this.positions[p].x + dx, y: this.positions[p].y + dy };
  }

  /** Brings pieces to the top, keeping their order among themselves. */
  private raise(ids: readonly number[]): void {
    const lifted = new Set(ids);
    this.order = [...this.order.filter((p) => !lifted.has(p)), ...this.order.filter((p) => lifted.has(p))];
  }
```

```ts
import type { KeyValueStore } from '../shared/store';
import { JIGSAW_STYLES, PIECE_COUNTS, type JigsawProgress, type JigsawStyle, type PieceCount } from './types';

export const JIGSAW_KEY = 'word-games:jigsaw:v1';
/** Her place is kept in this many picture / style / piece-count mixes, most recent first. */
export const MAX_PLACES = 20;

/** Her place in one mix. */
export interface SavedPlace {
  picture: string;
  style: JigsawStyle;
  count: PieceCount;
  progress: JigsawProgress;
}

export interface JigsawStats {
  totalSolved: number;
  todaySolved: number;
  lastPlayedDay: string | null;
}

/** Her style and piece count, the picture in progress, her places, the pictures she has finished, and her counts. */
export interface SavedJigsaw {
  style: JigsawStyle;
  count: PieceCount;
  /** The picture in progress, or null while she's choosing one. */
  current: string | null;
  places: SavedPlace[];
  finished: string[];
  stats: JigsawStats;
}

export function freshJigsaw(): SavedJigsaw {
  return { style: 'tray', count: 9, current: null, places: [], finished: [], stats: { totalSolved: 0, todaySolved: 0, lastPlayedDay: null } };
}

const DAY = /^\d{4}-\d{2}-\d{2}$/;
const isRecord = (value: unknown): value is Record<string, unknown> => typeof value === 'object' && value !== null;
const isCount = (value: unknown): value is number => Number.isInteger(value) && (value as number) >= 0;
const isNumber = (value: unknown): value is number => typeof value === 'number' && Number.isFinite(value);
const isCountList = (value: unknown): value is number[] => Array.isArray(value) && value.every(isCount);
const isTuple = (value: unknown, length: number) => Array.isArray(value) && value.length === length && value.every(isNumber);

/** The saved shape only; whether it fits its puzzle is JigsawGame.restore's job. */
function readProgress(v: unknown): JigsawProgress | null {
  if (!isRecord(v)) return null;
  const { placed, queue, tray, loose, positions, groups, order, hints, elapsedMs } = v;
  if (!isCountList(placed) || !isCountList(queue) || !isCountList(groups) || !isCountList(order)) return null;
  if (!Array.isArray(tray) || !tray.every((id) => id === null || isCount(id))) return null;
  if (!Array.isArray(loose) || !loose.every((entry) => isTuple(entry, 3) && isCount((entry as number[])[0]))) return null;
  if (!Array.isArray(positions) || !positions.every((entry) => isTuple(entry, 2))) return null;
  if (!isCount(hints) || !isNumber(elapsedMs) || elapsedMs < 0) return null;
  return {
    placed,
    queue,
    tray: tray as Array<number | null>,
    loose: loose as Array<[number, number, number]>,
    positions: positions as Array<[number, number]>,
    groups,
    order,
    hints,
    elapsedMs,
  };
}

function readPlace(v: unknown): SavedPlace | null {
  if (!isRecord(v) || typeof v.picture !== 'string') return null;
  if (!JIGSAW_STYLES.includes(v.style as JigsawStyle) || !PIECE_COUNTS.includes(v.count as PieceCount)) return null;
  const progress = readProgress(v.progress);
  return progress && { picture: v.picture, style: v.style as JigsawStyle, count: v.count as PieceCount, progress };
}

function readStats(v: unknown): JigsawStats {
  if (
    isRecord(v) &&
    isCount(v.totalSolved) &&
    isCount(v.todaySolved) &&
    (v.lastPlayedDay === null || (typeof v.lastPlayedDay === 'string' && DAY.test(v.lastPlayedDay)))
  ) {
    return { totalSolved: v.totalSolved, todaySolved: v.todaySolved, lastPlayedDay: v.lastPlayedDay };
  }
  return freshJigsaw().stats;
}

/** Loads the save field by field; anything damaged gets the fresh value. */
export function loadSavedJigsaw(store: KeyValueStore): SavedJigsaw {
  const saved = freshJigsaw();
  let value: unknown = null;
  try {
    const raw = store.getItem(JIGSAW_KEY);
    value = raw === null ? null : JSON.parse(raw);
  } catch {
    // Unreadable or unavailable: a fresh start.
  }
  if (!isRecord(value)) return saved;
  if (JIGSAW_STYLES.includes(value.style as JigsawStyle)) saved.style = value.style as JigsawStyle;
  if (PIECE_COUNTS.includes(value.count as PieceCount)) saved.count = value.count as PieceCount;
  if (typeof value.current === 'string') saved.current = value.current;
  if (Array.isArray(value.places)) saved.places = value.places.map(readPlace).filter((p): p is SavedPlace => p !== null).slice(0, MAX_PLACES);
  if (Array.isArray(value.finished)) saved.finished = [...new Set(value.finished.filter((id): id is string => typeof id === 'string'))];
  saved.stats = readStats(value.stats);
  return saved;
}

export function saveSavedJigsaw(store: KeyValueStore, saved: SavedJigsaw): void {
  try {
    store.setItem(JIGSAW_KEY, JSON.stringify(saved));
  } catch {
    // Storage full or unavailable: this session still works.
  }
}

const isMix = (place: SavedPlace, picture: string, style: JigsawStyle, count: PieceCount) =>
  place.picture === picture && place.style === style && place.count === count;

export function placeFor(saved: SavedJigsaw, picture: string, style: JigsawStyle, count: PieceCount): SavedPlace | undefined {
  return saved.places.find((place) => isMix(place, picture, style, count));
}

/** Her place in this mix, moved to the front (the oldest beyond 20 are forgotten). */
export function withPlace(saved: SavedJigsaw, place: SavedPlace): SavedJigsaw {
  const others = saved.places.filter((p) => !isMix(p, place.picture, place.style, place.count));
  return { ...saved, places: [place, ...others].slice(0, MAX_PLACES) };
}

export function withoutPlace(saved: SavedJigsaw, picture: string, style: JigsawStyle, count: PieceCount): SavedJigsaw {
  return { ...saved, places: saved.places.filter((p) => !isMix(p, picture, style, count)) };
}

/** Gives the picture its check on the picture page. */
export function withFinished(saved: SavedJigsaw, picture: string): SavedJigsaw {
  return saved.finished.includes(picture) ? saved : { ...saved, finished: [...saved.finished, picture] };
}

/** Counts a finished jigsaw; "today" starts over on a new day. */
export function recordJigsawSolve(stats: JigsawStats, day: string): JigsawStats {
  return {
    totalSolved: stats.totalSolved + 1,
    todaySolved: stats.lastPlayedDay === day ? stats.todaySolved + 1 : 1,
    lastPlayedDay: day,
  };
}
```

```ts
import { piecePath, tabFor, type PieceShape } from './cut';

/** Sharper pieces on Retina screens, but never more than twice the size, to save memory. */
const MAX_SCALE = 2;

/**
 * Cuts a 4:3 picture into piece images, each w × h plus its tabs: the photo inside the piece's outline, with a thin
 * white edge so neighbors stand apart until the end.
 */
export function drawPieces(image: HTMLImageElement, shapes: readonly PieceShape[], cols: number, rows: number, w: number, h: number): HTMLCanvasElement[] {
  const tab = tabFor(w, h);
  const scale = Math.min(MAX_SCALE, window.devicePixelRatio || 1);
  // Picture pixels per screen pixel.
  const sx = image.naturalWidth / (cols * w);
  const sy = image.naturalHeight / (rows * h);
  return shapes.map((shape) => {
    const canvas = document.createElement('canvas');
    canvas.width = Math.ceil((w + 2 * tab) * scale);
    canvas.height = Math.ceil((h + 2 * tab) * scale);
    const context = canvas.getContext('2d');
    if (!context) return canvas;
    context.scale(scale, scale);
    const outline = new Path2D(piecePath(shape, w, h, tab));
    context.save();
    context.clip(outline);
    // The piece's box in the picture, kept inside the picture: older Safari draws nothing for a box that isn't.
    const left = shape.col * w - tab;
    const top = shape.row * h - tab;
    const [x0, y0] = [Math.max(0, left), Math.max(0, top)];
    const [x1, y1] = [Math.min(cols * w, left + w + 2 * tab), Math.min(rows * h, top + h + 2 * tab)];
    context.drawImage(image, x0 * sx, y0 * sy, (x1 - x0) * sx, (y1 - y0) * sy, x0 - left, y0 - top, x1 - x0, y1 - y0);
    context.restore();
    context.strokeStyle = 'rgba(255, 255, 255, 0.9)';
    context.lineWidth = 2;
    context.stroke(outline);
    return canvas;
  });
}
```

```ts
import type { Rect } from '../frame/layout';
import { el, place, restartAnimation, wait } from '../ui/dom';
import { flyHintStar, landHintLetter } from '../ui/hint-star';
import { CHECK_ICON, HINT_ICON, PICTURES_ICON } from '../ui/icons';
import { piecePath, tabFor, type PieceShape } from './cut';
import type { JigsawLayout } from './layout';
import { drawPieces } from './pieces';
import { JIGSAW_STYLES, PIECE_COUNTS, type JigsawStyle, type Picture, type PieceCount } from './types';

const STYLE_LABELS: Record<JigsawStyle, string> = { tray: 'Tray', board: 'Board', table: 'Table' };
/** At the end the seams fade into the whole photo (matches the CSS)... */
const FADE_MS = 600;
/** ...which then grows into place. */
const GROW_MS = 500;
const SVG_NS = 'http://www.w3.org/2000/svg';

/** Where one piece is drawn: its body (without tabs) on screen, and what it's doing. */
export interface PieceSpot {
  id: number;
  body: Rect;
  where: 'tray' | 'placed' | 'loose' | 'table';
  /** The tray place it sits in, shrunk to fit (where = 'tray'). */
  slot?: Rect;
}

export interface JigsawViewState {
  style: JigsawStyle;
  count: PieceCount;
  picture: Picture | null;
  /** The loaded picture, or null while it loads (then no pieces are drawn). */
  image: HTMLImageElement | null;
  shapes: readonly PieceShape[];
  /** A piece's body size: board-sized, or smaller on the table. */
  pieceSize: { w: number; h: number };
  /** Every piece, bottom first. */
  pieces: readonly PieceSpot[];
  /** The piece lifted by a tap, waiting for a tap on its spot. */
  lifted: number | null;
}

const outer = (body: Rect, tab: number): Rect => ({ x: body.x - tab, y: body.y - tab, w: body.w + 2 * tab, h: body.h + 2 * tab });
const WHERE = ['tray', 'placed', 'loose', 'table'] as const;

/** Draws Jigsaw and plays its animations. Knows nothing about rules. */
export class JigsawView {
  private layout!: JigsawLayout;
  private board: HTMLElement | null = null;
  private hintButton: HTMLElement | null = null;
  private readonly pieces = new Map<number, HTMLElement>();
  private canvases: HTMLCanvasElement[] = [];
  private cutKey = '';
  private tab = 0;

  /** Buttons and pieces are drawn here but touched through the touch tracker, which works with a resting thumb. */
  constructor(private readonly layer: HTMLElement) {}

  render(layout: JigsawLayout, state: JigsawViewState, animateIn: boolean): void {
    this.layout = layout;
    const board = document.createElement('div');
    board.className = `board jigsaw ${state.style}`;
    this.layer.replaceChildren(board);
    this.board = board;

    this.hintButton = this.button(board, 'button hint', `${HINT_ICON}<span>Hint</span>`, layout.hintButton);
    this.button(board, 'button pictures', `${PICTURES_ICON}<span>Pictures</span>`, layout.picturesButton);
    JIGSAW_STYLES.forEach((style, i) => this.choice(board, STYLE_LABELS[style], style === state.style, layout.styleButtons[i]));
    PIECE_COUNTS.forEach((count, i) => this.choice(board, String(count), count === state.count, layout.countButtons[i]));
    place(el('div', 'tray', board), layout.tray);
    if (state.style !== 'table') this.drawBoard(board, state);
    else if (state.picture) {
      const lid = el('img', 'jigsaw-lid', board);
      lid.alt = '';
      lid.src = state.picture.src;
      place(lid, layout.lid);
    }

    this.cut(state);
    this.pieces.clear();
    for (const spot of state.pieces) {
      const piece = el('div', 'piece', board);
      const canvas = this.canvases[spot.id];
      if (canvas) piece.appendChild(canvas);
      this.pieces.set(spot.id, piece);
    }
    this.update(state);
    if (animateIn) {
      board.classList.add('entering');
      void board.offsetWidth;
      board.classList.remove('entering');
    }
  }

  /** Puts every piece where it is now; pieces that moved glide there (the CSS transition). Held pieces are left alone. */
  update(state: JigsawViewState): void {
    state.pieces.forEach((spot, i) => {
      const piece = this.pieces.get(spot.id);
      if (!piece || piece.classList.contains('held')) return;
      const rect = outer(spot.body, this.tab);
      place(piece, rect);
      // Stacked with z-index rather than page order: moving an element in the page would stop its glide.
      piece.style.zIndex = String(i + 1);
      piece.style.transform = spot.slot ? `scale(${Math.min(spot.slot.w / rect.w, spot.slot.h / rect.h)})` : '';
      for (const where of WHERE) piece.classList.toggle(where, where === spot.where);
      piece.classList.toggle('lifted', state.lifted === spot.id);
    });
  }

  /** She picked these pieces up: full size, above everything, following her finger. */
  hold(ids: readonly number[]): void {
    for (const id of ids) {
      const piece = this.pieces.get(id);
      if (!piece) continue;
      piece.classList.add('held');
      piece.classList.remove('lifted');
      piece.style.transform = '';
    }
  }

  moveHeld(spots: ReadonlyArray<{ id: number; body: Rect }>): void {
    for (const { id, body } of spots) {
      const piece = this.pieces.get(id);
      if (piece) place(piece, outer(body, this.tab));
    }
  }

  /** She let go: the next update glides them to wherever they belong. */
  release(): void {
    for (const piece of this.pieces.values()) piece.classList.remove('held');
  }

  /** A piece clicks in (or joins others). */
  pop(id: number): void {
    const piece = this.pieces.get(id);
    if (piece) restartAnimation(piece, 'pop');
  }

  /** A table piece she tapped: a little wiggle shows it can be moved. */
  wiggle(id: number): void {
    const piece = this.pieces.get(id);
    if (piece) restartAnimation(piece, 'wiggle');
  }

  /** The hint star flies from Hint to `target`. Resolves when it lands. */
  async flyStar(target: Rect): Promise<void> {
    const board = this.board;
    const hint = this.hintButton;
    if (!board || !hint) return;
    const spot = el('div', 'jigsaw-star-target', board);
    place(spot, target);
    await flyHintStar(hint, spot);
    spot.remove();
  }

  /** The hinted piece lands with a bounce. */
  land(id: number): void {
    const piece = this.pieces.get(id);
    if (piece) landHintLetter(piece);
  }

  setHintPulse(on: boolean): void {
    this.hintButton?.classList.toggle('pulse', on);
  }

  /** The seams fade into the whole photo, which grows into place with its caption under it. */
  async finish(picture: Picture, assembled: Rect): Promise<void> {
    const board = this.board;
    if (!board) return;
    const photo = el('img', 'jigsaw-photo', board);
    photo.alt = '';
    photo.src = picture.src;
    place(photo, assembled);
    void photo.offsetWidth;
    photo.classList.add('shown');
    await wait(FADE_MS);
    for (const piece of this.pieces.values()) piece.remove();
    board.classList.add('done');
    place(photo, this.layout.finished);
    if (picture.caption) {
      const caption = el('div', 'jigsaw-caption', board);
      caption.textContent = picture.caption;
      place(caption, this.layout.caption);
    }
    await wait(GROW_MS);
  }

  private button(parent: HTMLElement, className: string, html: string, rect: Rect): HTMLButtonElement {
    const button = el('button', className, parent);
    button.type = 'button';
    button.innerHTML = html;
    place(button, rect);
    return button;
  }

  /** A style or piece-count button, in the level buttons' look. */
  private choice(parent: HTMLElement, label: string, on: boolean, rect: Rect): void {
    this.button(parent, on ? 'level on' : 'level', `${on ? CHECK_ICON : ''}<span>${label}</span>`, rect);
  }

  /** The board: the picture faintly, with thin lines along the piece edges. */
  private drawBoard(board: HTMLElement, state: JigsawViewState): void {
    const back = el('div', 'jigsaw-board', board);
    place(back, this.layout.board);
    if (!state.picture) return;
    const faint = el('img', 'jigsaw-faint', back);
    faint.alt = '';
    faint.src = state.picture.src;
    if (state.shapes.length === 0) return;
    const { w, h } = state.pieceSize;
    const tab = tabFor(w, h);
    const lines = document.createElementNS(SVG_NS, 'svg');
    lines.setAttribute('class', 'jigsaw-lines');
    for (const shape of state.shapes) {
      const path = document.createElementNS(SVG_NS, 'path');
      path.setAttribute('d', piecePath(shape, w, h, tab));
      path.setAttribute('transform', `translate(${shape.col * w - tab} ${shape.row * h - tab})`);
      lines.appendChild(path);
    }
    back.appendChild(lines);
  }

  /** Cuts the picture into piece images once per picture, piece count and size. */
  private cut(state: JigsawViewState): void {
    const { image, picture, shapes, pieceSize } = state;
    this.tab = tabFor(pieceSize.w, pieceSize.h);
    if (!image || !picture || shapes.length === 0) {
      this.canvases = [];
      this.cutKey = '';
      return;
    }
    const key = `${picture.src}|${shapes.length}|${pieceSize.w.toFixed(2)}x${pieceSize.h.toFixed(2)}`;
    if (key === this.cutKey) return;
    const cols = Math.max(...shapes.map((s) => s.col)) + 1;
    const rows = Math.max(...shapes.map((s) => s.row)) + 1;
    this.canvases = drawPieces(image, shapes, cols, rows, pieceSize.w, pieceSize.h);
    this.cutKey = key;
  }
}
```

```ts
    place(tile, layout.tiles[i]);
    const image = el('img', '', tile);
    image.alt = '';
    image.decoding = 'async';
    image.src = picture.thumb;
    if (state.finished.has(picture.id)) el('span', 'pic-check', tile).innerHTML = CHECK_ICON;
    if (picture.caption) {
      const caption = el('div', 'pic-caption', page);
      caption.textContent = picture.caption;
      place(caption, layout.captions[i]);
    }
  });
  if (state.hasBack) button(page, 'button pics-back', '‹ Back', layout.back);
  if (state.canClose) button(page, 'button pics-close', 'Close', layout.close);
  if (state.hasMore) button(page, 'button pics-more', 'More pictures ›', layout.more);

  if (animateIn) {
    page.classList.add('entering');
    void page.offsetWidth;
    page.classList.remove('entering');
  }
```

```ts
/** What she's holding: pieces that move together (a Table group), where they started, and where her finger is on the first. */
interface Held {
  id: number;
  ids: number[];
  starts: Rect[];
  /** Set on the first move. */
  grab: Point | null;
  fromTray: boolean;
}

export interface JigsawOptions {

const boundsOf = (rects: readonly Rect[]): Rect => {
  const x = Math.min(...rects.map((r) => r.x));
  const y = Math.min(...rects.map((r) => r.y));
  return { x, y, w: Math.max(...rects.map((r) => r.x + r.w)) - x, h: Math.max(...rects.map((r) => r.y + r.h)) - y };
};

const moved = (a: PieceSpot, b: PieceSpot) => a.where !== b.where || a.body.x !== b.body.x || a.body.y !== b.body.y;

/** Jigsaw: big-piece puzzles from pictures, played from a tray, on a board, or loose on a table. */
export class JigsawScreen implements GameScreen {
  readonly id = 'jigsaw';
  readonly title = GAME_TITLES.jigsaw;
  /** True while the picture is coming together or she's celebrating. */
  busy = false;
  private host!: FrameHost;
  private saved!: SavedJigsaw;
  private mode: 'pictures' | 'puzzle' = 'pictures';
  /** Null until the list has loaded. */
  private pictures: Picture[] | null = null;
  /** The picture in progress. */
  private picture: Picture | null = null;
  private page = 0;
  /** Made at the next render (Table pieces start where the layout has room). */
  private game: JigsawGame | null = null;
  private shapes: PieceShape[] = [];
  private readonly images = new Map<string, HTMLImageElement>();
  private held: Held | null = null;
  private lifted: number | null = null;
  private hintFlying = false;
  /** While the hint star flies, the pieces it moves are drawn where they were. */
  private frozen: Map<number, PieceSpot> | null = null;
  /** True while Jigsaw is on screen (between resume and pause). */
  private shown = false;
  private view: JigsawView | null = null;
  private layout: JigsawLayout | null = null;
  private idleTimer = 0;
  private timer = new PlayTimer();
  private readonly now: () => Date;

  constructor(
    private readonly store: KeyValueStore,
    private readonly options: JigsawOptions,
  ) {
    this.now = options.now ?? (() => new Date());
  }

  start(host: FrameHost): void {
    this.host = host;
    this.saved = loadSavedJigsaw(this.store);
    this.mode = this.saved.current ? 'puzzle' : 'pictures';
    this.timer.pause();
    void this.options.loadPictures().then((pictures) => this.picturesLoaded(pictures));
  }

  render(layer: HTMLElement, frame: FrameLayout, animateIn: boolean): GameTargets {
    if (!this.view) this.view = new JigsawView(layer);
    this.held = null;
    this.layout = computeJigsawLayout(frame);
    if (this.mode === 'pictures') return this.renderPictures(layer, frame, animateIn);
    this.ensureGame(this.layout);
    this.view.render(this.layout, this.viewState(), animateIn);
    return this.targets();
  }

  buttonTapped(id: string): void {
    if (this.busy) return;
    if (id.startsWith('pic:')) this.openPicture(Number(id.slice('pic:'.length)));
    else if (id === 'pics-back' || id === 'pics-more') this.turnPage(id === 'pics-more' ? 1 : -1);
    else if (id === 'pics-close') this.closePictures();
    else if (id === 'pictures') this.showPictures();
    else if (id === 'hint') void this.hint();
    else if (id.startsWith('style:')) this.choose({ style: id.slice('style:'.length) as JigsawStyle });
    else if (id.startsWith('count:')) this.choose({ count: Number(id.slice('count:'.length)) as PieceCount });
    else if (id.startsWith('piece:')) this.tapPiece(Number(id.slice('piece:'.length)));
    else if (id.startsWith('spot:')) this.tapSpot(Number(id.slice('spot:'.length)));
  }

  /** She picked a piece up. Its targets wait until she lets go: new targets now would cancel the drag. */
  dragStarted(id: string): void {
    const game = this.game;
    if (this.busy || !game || !this.view || this.mode !== 'puzzle' || !id.startsWith('piece:')) return;
    const piece = Number(id.slice('piece:'.length));
    const spots = this.pieceSpots();
    // A table piece brings its group; the one under her finger comes first.
    const ids = [piece, ...game.groupOf(piece).filter((p) => p !== piece)];
    const starts = ids.map((p) => spots.find((s) => s.id === p)?.body);
    if (starts.some((s) => !s)) return;
    this.lifted = null;
    this.held = { id: piece, ids, starts: starts as Rect[], grab: null, fromTray: game.tray.includes(piece) };
    this.view.hold(ids);
  }

  dragMoved(_id: string, x: number, y: number): void {
    const held = this.held;
    if (!held || !this.view) return;
    if (!held.grab) {
      const first = held.starts[0];
      // A piece from the tray grows to full size under her finger; anything else keeps its place under her finger.
      if (held.fromTray) {
        held.starts[0] = { ...first, x: x - first.w / 2, y: y - first.h / 2 };
        held.grab = { x: first.w / 2, y: first.h / 2 };
      } else {
        held.grab = { x: x - first.x, y: y - first.y };
      }
    }
    this.view.moveHeld(this.heldBodies(held, x, y));
  }

  dragEnded(_id: string, x: number, y: number): void {
    const held = this.held;
    const game = this.game;
    const layout = this.layout;
    this.held = null;
    if (!held?.grab || !game || !layout || !this.view) return this.dragCancelled();
    const bodies = this.heldBodies(held, x, y).map((b) => b.body);
    // What she drops stays inside the free area (a group as a whole).
    const box = boundsOf(bodies);
    const kept = clampToFree(layout.free, box);
    const first = { ...bodies[0], x: bodies[0].x + kept.x - box.x, y: bodies[0].y + kept.y - box.y };
    const result = game.drop(held.id, this.toPieces(layout, game, first));
    this.view.release();
    this.changed();
    if (result === 'placed' || result === 'joined') this.view.pop(held.id);
  }

  /** The iPad cancelled the touch (or the screen changed): the pieces glide back. Targets are unchanged. */
  dragCancelled(): void {
    this.held = null;
    this.view?.release();
    if (this.mode === 'puzzle' && this.game) this.view?.update(this.viewState());
  }

  pause(): void {
    this.shown = false;
    this.timer.pause();
    window.clearTimeout(this.idleTimer);
    this.view?.setHintPulse(false);
    if (this.held) this.dragCancelled();
    this.save();
  }

  resume(): void {
    this.shown = true;
    if (this.mode === 'puzzle') this.timer.resume(performance.now());
    // A puzzle finished just before she switched away (by a piece while a hint star flew) is celebrated when she's back.
    if (this.game?.solved && !this.busy && !this.hintFlying) void this.finish();
    else if (!this.busy) this.resetIdle();
  }

  activity(): void {
    if (this.mode === 'puzzle') this.timer.activity(performance.now());
  }

  summary(): GameSummary {
    return { solved: this.saved.stats.totalSolved };
  }

  private picturesLoaded(pictures: Picture[]): void {
    this.pictures = pictures;
    const current = this.saved.current;
    if (current && !this.picture) {

      // Its picture is gone from the site: back to choosing.
      if (!this.picture) {
        this.saved = { ...this.saved, current: null };
        this.mode = 'pictures';
        saveSavedJigsaw(this.store, this.saved);
      }
    }
    if (this.shown && !this.busy && !this.held) this.host.redraw(false);
  }

  private renderPictures(layer: HTMLElement, frame: FrameLayout, animateIn: boolean): GameTargets {
    const layout = computePicturePageLayout(frame);
    const page = picturePage(this.pictures ?? [], this.page);
    renderPicturePage(

    if (page.hasBack) buttons.push({ id: 'pics-back', ...layout.back });
    if (this.picture) buttons.push({ id: 'pics-close', ...layout.close });
    if (page.hasMore) buttons.push({ id: 'pics-more', ...layout.more });
    return { tiles: [], buttons };
  }

  /** The puzzle for her picture, style and piece count: her saved place in it, or a new one. */
  private ensureGame(layout: JigsawLayout): void {
    const picture = this.picture;
    if (this.game || !picture) return;
    const { style, count } = this.saved;
    const { cols, rows } = GRID_SIZES[count];
    const seed = pieceSeed(picture.id, count);
    const place = placeFor(this.saved, picture.id, style, count);
    const restored = place ? JigsawGame.restore(style, count, place.progress) : null;
    this.game = restored ?? JigsawGame.fresh(style, count, seed, style === 'table' ? this.startSpots(layout, count) : []);
    this.shapes = cutPieces(cols, rows, seed);
    this.timer = new PlayTimer(restored ? place!.progress.elapsedMs : 0);
    if (document.hidden) this.timer.pause();
    else this.timer.activity(performance.now());
    this.save();
  }

  /** Where table pieces may start, in table pieces from the table's corner. */
  private startSpots(layout: JigsawLayout, count: PieceCount): Point[] {
    const table = tablePieces(layout, count);
    const origin = layout.free[0];
    return table.spots.map((s) => ({ x: (s.x - origin.x) / table.w, y: (s.y - origin.y) / table.h }));
  }

  private save(): void {
    const { game, picture } = this;
    if (game && picture && !game.solved) {
      this.saved = withPlace(this.saved, { picture: picture.id, style: game.style, count: game.count, progress: game.progress(this.timer.elapsedMs) });
    }
    saveSavedJigsaw(this.store, this.saved);
  }

  private imageFor(picture: Picture): HTMLImageElement {
    let image = this.images.get(picture.src);
    if (!image) {
      image = new Image();
      image.decoding = 'async';
      image.onload = () => {
        if (this.picture?.src === picture.src && this.mode === 'puzzle' && this.shown && !this.busy && !this.held) this.host.redraw(false);
      };
      image.src = picture.src;
      this.images.set(picture.src, image);
    }
    return image;
  }

  private ready(): HTMLImageElement | null {
    const image = this.picture && this.imageFor(this.picture);
    return image && image.complete && image.naturalWidth > 0 ? image : null;
  }

  private boardSize(layout: JigsawLayout, count: PieceCount): { w: number; h: number } {
    const { cols, rows } = GRID_SIZES[count];
    return { w: layout.board.w / cols, h: layout.board.h / rows };
  }

  private pieceSize(layout: JigsawLayout, game: JigsawGame): { w: number; h: number } {
    if (game.style !== 'table') return this.boardSize(layout, game.count);
    const { w, h } = tablePieces(layout, game.count);
    return { w, h };
  }

  /** Positions are counted from the board's corner (Tray, Board) or the table's (Table). */
  private origin(layout: JigsawLayout, game: JigsawGame): Point {
    return game.style === 'table' ? layout.free[0] : layout.board;
  }

  private toPieces(layout: JigsawLayout, game: JigsawGame, body: Rect): Point {
    const { w, h } = this.pieceSize(layout, game);
    const origin = this.origin(layout, game);
    return { x: (body.x - origin.x) / w, y: (body.y - origin.y) / h };
  }

  private fromPieces(layout: JigsawLayout, game: JigsawGame, at: Point): Rect {
    const { w, h } = this.pieceSize(layout, game);
    const origin = this.origin(layout, game);
    return { x: origin.x + at.x * w, y: origin.y + at.y * h, w, h };
  }

  /** Where every piece is drawn, bottom first. */
  private pieceSpots(): PieceSpot[] {
    const { game, layout } = this;
    if (!game || !layout) return [];
    let spots: PieceSpot[];
    if (game.style === 'table') {
      const bodies = new Map(game.stacking.map((id) => [id, this.fromPieces(layout, game, game.positionOf(id))]));
      // Each group is kept inside the free area as a whole (the screen may have changed since it was put down).
      const done = new Set<number>();
      for (const id of game.stacking) {
        if (done.has(id)) continue;
        const members = game.groupOf(id);
        members.forEach((m) => done.add(m));
        const box = boundsOf(members.map((m) => bodies.get(m)!));
        const kept = clampToFree(layout.free, box);
        for (const m of members) {
          const b = bodies.get(m)!;
          bodies.set(m, { ...b, x: b.x + kept.x - box.x, y: b.y + kept.y - box.y });
        }
      }
      spots = game.stacking.map((id) => ({ id, body: bodies.get(id)!, where: 'table' }));
    } else {
      const { w, h } = this.boardSize(layout, game.count);
      spots = game.placed.map((id) => ({ id, body: this.fromPieces(layout, game, game.home(id)), where: 'placed' as const }));
      for (const [id, at] of game.loose) spots.push({ id, body: clampToFree(layout.free, this.fromPieces(layout, game, at)), where: 'loose' });
      game.tray.forEach((id, i) => {
        if (id === null) return;
        const slot = layout.slots[i];
        spots.push({ id, body: { x: slot.x + slot.w / 2 - w / 2, y: slot.y + slot.h / 2 - h / 2, w, h }, where: 'tray', slot });
      });
    }
    const frozen = this.frozen;
    return frozen ? spots.map((s) => frozen.get(s.id) ?? s) : spots;
  }

  private viewState(): JigsawViewState {
    const { game, layout } = this;
    const image = this.ready();
    return {
      style: this.saved.style,
      count: this.saved.count,
      picture: this.picture,
      image,
      shapes: image && game ? this.shapes : [],
      pieceSize: game && layout ? this.pieceSize(layout, game) : { w: 0, h: 0 },
      pieces: image ? this.pieceSpots() : [],
      lifted: this.lifted,
    };
  }

  private targets(): GameTargets {
    const { game, layout } = this;
    if (!layout) return { tiles: [], buttons: [] };
    const buttons: ButtonGeometry[] = [
      { id: 'hint', ...layout.hintButton },
      { id: 'pictures', ...layout.picturesButton },
      ...JIGSAW_STYLES.map((style, i) => ({ id: `style:${style}`, ...layout.styleButtons[i] })),
      ...PIECE_COUNTS.map((count, i) => ({ id: `count:${count}`, ...layout.countButtons[i] })),
    ];
    const draggables: ButtonGeometry[] = [];
    if (game && this.ready()) {
      // Topmost first: the touch tracker takes the first piece under her finger.
      for (const spot of [...this.pieceSpots()].reverse()) {
        if (spot.where === 'placed' || this.frozen?.has(spot.id)) continue;
        draggables.push({ id: `piece:${spot.id}`, ...(spot.slot ?? spot.body) });
      }
      // With a piece lifted by a tap, the board's empty spots can be tapped.
      if (this.lifted !== null) {
        for (let id = 0; id < game.count; id++) if (!game.isPlaced(id)) buttons.push({ id: `spot:${id}`, ...this.fromPieces(layout, game, game.home(id)) });
      }
    }
    return { tiles: [], buttons, draggables, reviewButton: layout.reviewButton };
  }

  private heldBodies(held: Held, x: number, y: number): Array<{ id: number; body: Rect }> {
    const dx = x - held.grab!.x - held.starts[0].x;
    const dy = y - held.grab!.y - held.starts[0].y;
    return held.ids.map((id, i) => ({ id, body: { ...held.starts[i], x: held.starts[i].x + dx, y: held.starts[i].y + dy } }));
  }

  /** Redraws the pieces and tells the touch tracker what can be touched now. */
  private refresh(): void {
    if (this.mode !== 'puzzle') return;
    this.view?.update(this.viewState());
    this.host.updateTargets(this.targets());
  }

  /** Something moved: save it, show it, and see if she's finished. */
  private changed(): void {
    this.save();
    this.resetIdle();
    this.refresh();
    // If the hint star is still flying, it finishes the puzzle when it lands.
    if (this.game?.solved && !this.hintFlying && !this.busy) void this.finish();
  }

  /** Tray and Board: a tap lifts a piece (or puts it down again). Table: a wiggle. */
  private tapPiece(id: number): void {
    const game = this.game;
    if (!game || this.mode !== 'puzzle') return;
    if (game.style === 'table') return this.view?.wiggle(id);
    this.lifted = this.lifted === id ? null : id;
    this.refresh();
  }

  /** A tap on the board with a piece lifted: its own spot clicks it in; Board leaves it on any other spot. */
  private tapSpot(spot: number): void {
    const game = this.game;
    const id = this.lifted;
    if (!game || id === null) return;
    this.lifted = null;
    const result = game.drop(id, game.home(spot));
    this.changed();
    if (result === 'placed') this.view?.pop(id);
  }

  private async hint(): Promise<void> {
    const game = this.game;
    // One star at a time, so a double tap doesn't use two hints.
    if (this.hintFlying || !game || this.mode !== 'puzzle' || !this.ready() || game.solved) return;
    this.lifted = null;
    const before = new Map(this.pieceSpots().map((s) => [s.id, s]));
    const id = game.hint();
    if (id === null) return;
    const after = this.pieceSpots();
    const target = after.find((s) => s.id === id)!.body;
    this.frozen = new Map(after.filter((s) => before.has(s.id) && moved(before.get(s.id)!, s)).map((s) => [s.id, before.get(s.id)!]));
    this.hintFlying = true;
    // A hint that finishes the puzzle: nothing more to do until "Well done!".
    if (game.solved) this.busy = true;
    this.save();
    this.resetIdle();
    this.refresh();
    await this.view?.flyStar(target);
    this.hintFlying = false;
    this.frozen = null;
    // If she chose another picture, style or piece count while the star flew, that puzzle is gone from the screen.
    if (this.game === game && this.mode === 'puzzle') {
      this.view?.update(this.viewState());
      if (this.shown) this.view?.land(id);
      // Setting the touch targets while she holds a piece would drop it; they're set when she lets go.
      if (!this.held) this.host.updateTargets(this.targets());
    }
    if (game.solved && this.game === game && (this.busy || this.shown)) void this.finish();
  }

  /** Each picture, style and piece count keeps her place, so switching never loses anything. */
  private choose(change: { style?: JigsawStyle; count?: PieceCount }): void {
    const style = change.style ?? this.saved.style;
    const count = change.count ?? this.saved.count;
    if (!JIGSAW_STYLES.includes(style) || !PIECE_COUNTS.includes(count)) return;
    if (style === this.saved.style && count === this.saved.count) return;
    this.save();
    this.saved = { ...this.saved, style, count };
    this.game = null;
    this.lifted = null;
    saveSavedJigsaw(this.store, this.saved);
    this.host.redraw(true);
    this.resetIdle();
  }

  private openPicture(index: number): void {
    const picture = picturePage(this.pictures ?? [], this.page).pictures[index];
    if (!picture) return;
    this.save();
    this.picture = picture;
    this.game = null;
    this.lifted = null;
    // Only the picture on screen is kept decoded: a 1600 × 1200 photo takes several MB.
    for (const src of [...this.images.keys()]) if (src !== picture.src) this.images.delete(src);
    this.saved = { ...this.saved, current: picture.id };
    this.mode = 'puzzle';
    this.host.redraw(true);
    this.timer.resume(performance.now());
    this.resetIdle();
  }

  private showPictures(): void {
    this.save();
    this.lifted = null;
    this.timer.pause();
    window.clearTimeout(this.idleTimer);
    this.view?.setHintPulse(false);
    this.mode = 'pictures';
    this.page = 0;
    this.host.redraw(true);
  }

  private closePictures(): void {
    if (!this.picture) return;
    this.mode = 'puzzle';
    this.host.redraw(true);
    this.timer.resume(performance.now());
    this.resetIdle();
  }

  private turnPage(step: number): void {
    this.page = Math.max(0, this.page + step);
    this.host.redraw(false);
  }

  /** Where the finished picture is when it comes together: the board, or wherever the table's one group lies. */
  private assembled(layout: JigsawLayout): Rect {
    if (this.game?.style !== 'table') return layout.board;
    return boundsOf(this.pieceSpots().map((s) => s.body));
  }

  private async finish(): Promise<void> {
    const { game, picture, layout } = this;
    if (!game || !picture || !layout) return;
    // A piece she's still holding (picked up while a hint star flew) is let go: the puzzle is done.
    if (this.held) this.dragCancelled();
    this.busy = true;
    this.lifted = null;
    window.clearTimeout(this.idleTimer);
    this.view?.setHintPulse(false);
    if (this.shown) await this.view?.finish(picture, this.assembled(layout));
    this.timer.activity(performance.now());
    this.host.logPlay(jigsawRecord({ count: game.count, style: game.style, hints: game.hints, ms: this.timer.elapsedMs, now: this.now() }));
    const done = withFinished(withoutPlace(this.saved, picture.id, game.style, game.count), picture.id);
    this.saved = { ...done, current: null, stats: recordJigsawSolve(this.saved.stats, dayKey(this.now())) };
    const streakDays = this.host.recordPlayDay();
    // Saved as done now, so closing the app mid-celebration loses nothing.
    this.game = null;
    this.picture = null;
    this.timer.pause();
    saveSavedJigsaw(this.store, this.saved);
    if (this.shown) await wait(ADMIRE_MS);
    await this.host.celebrate({
      lines: jigsawLines({
        hinted: game.hints > 0,
        streakDays,
        todaySolved: this.saved.stats.todaySolved,
        puzzleNumber: this.saved.stats.totalSolved,
      }),
      nextLabel: 'Next picture',
      timeoutMs: CELEBRATION_MS,
    });
    this.mode = 'pictures';
    this.page = 0;
    this.busy = false;
    this.host.redraw(true);
  }

  private resetIdle(): void {
    window.clearTimeout(this.idleTimer);
    this.view?.setHintPulse(false);
    if (this.mode !== 'puzzle') return;
    this.idleTimer = window.setTimeout(() => this.view?.setHintPulse(true), IDLE_HINT_MS);
  }
}
```
