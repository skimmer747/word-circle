# Stats Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A play log of every finished puzzle or word, her lively good-news stats pop-up (opened from Games), and an honest "How she's doing" section on the caregiver settings page.

**Architecture:** Each game builds a `PlayRecord` when she finishes and the frame saves it (`src/stats/log.ts`, with lifetime figures that outlive the 400-day detail). Pure modules turn the log and the games' saved totals into her five cards (`her-stats.ts`) and the family's weekly figures (`family-stats.ts`); thin DOM/SVG views draw them (`her-view.ts`, `family-view.ts`).

**Tech Stack:** TypeScript, Vite, vitest (node environment, no DOM). Views are checked in the browser preview.

/** `day` moved by `n` days (negative for earlier), as "YYYY-MM-DD". */
export function addDays(day: string, n: number): string {
  const [y, m, d] = day.split('-').map(Number);
  return new Date(Date.UTC(y, m - 1, d + n)).toISOString().slice(0, 10);

```ts
import { GAME_IDS, type GameId } from '../frame/games';
import { addDays } from '../shared/days';
import type { KeyValueStore } from '../shared/store';

export const LOG_KEY = 'word-games:log:v1';
export const LIFETIME_KEY = 'word-games:lifetime:v1';
/** Detail older than this is dropped; the lifetime figures keep what lasts. */
export const KEEP_DAYS = 400;

export const PLAY_LEVELS = ['easy', 'medium', 'hard'] as const;
export type PlayLevel = (typeof PLAY_LEVELS)[number];

/** One finished puzzle or word. */
export interface PlayRecord {
  /** Local date, "2026-10-20". */
  day: string;
  /** Local hour she finished, 0–23. */
  hour: number;
  game: GameId;
  /** Null for Word Guess, which has no levels. */
  level: PlayLevel | null;
  /** Active play time. */
  ms: number;
  /** Letters revealed by Hint. */
  hints: number;
  /** Word Guess guesses; 0 for the other games. */
  tries: number;
  /** Word Guess: she typed the answer herself. Always true for the other games. */
  typed: boolean;
  /** Word Guess: it was today's Wordle. */
  daily: boolean;
  /** Words found in it, and the letters in them. */
  words: number;
  letters: number;
  /** The longest of those words. */
  longest: string;
  /** Word Guess: her first guess ('' if none); '' for the other games. */
  first: string;
}

/** Running figures since the log began; they stay when old detail is dropped. */
export interface Lifetime {
  /** The first logged day. */
  since: string | null;
  daysPlayed: number;
  lastDay: string | null;
  words: number;
  letters: number;
  longestWord: string;
  /** Word Guess, typed solves only. */
  fewestTries: number | null;
  /** Word Guess typed solves by tries: index 0 is 1 try … 5 is 6 tries, 6 is 7 or more. */
  triesCounts: number[];
  noHintGrids: number;
  firstGuesses: Record<string, number>;
  /** Records by the hour they were finished, 0–23. */
  hours: number[];
  mostInADay: number;
  /** Records on lastDay, for mostInADay. */
  todayCount: number;
}

export function emptyLifetime(): Lifetime {
  return {
    since: null,
    daysPlayed: 0,
    lastDay: null,
    words: 0,
    letters: 0,
    longestWord: '',
    fewestTries: null,
    triesCounts: [0, 0, 0, 0, 0, 0, 0],
    noHintGrids: 0,
    firstGuesses: {},
    hours: new Array<number>(24).fill(0),
    mostInADay: 0,
    todayCount: 0,
  };
}

const DAY = /^\d{4}-\d{2}-\d{2}$/;
const isRecord = (v: unknown): v is Record<string, unknown> => typeof v === 'object' && v !== null;
const isCount = (v: unknown): boolean => Number.isInteger(v) && (v as number) >= 0;
const isDay = (v: unknown): v is string => typeof v === 'string' && DAY.test(v);
const isCounts = (v: unknown, length: number): boolean => Array.isArray(v) && v.length === length && v.every(isCount);

function isPlayRecord(v: unknown): v is PlayRecord {
  return (
    isRecord(v) &&
    isDay(v.day) &&
    isCount(v.hour) &&
    (v.hour as number) < 24 &&
    GAME_IDS.includes(v.game as GameId) &&
    (v.level === null || PLAY_LEVELS.includes(v.level as PlayLevel)) &&
    typeof v.ms === 'number' &&
    Number.isFinite(v.ms) &&
    v.ms >= 0 &&
    isCount(v.hints) &&
    isCount(v.tries) &&
    typeof v.typed === 'boolean' &&
    typeof v.daily === 'boolean' &&
    isCount(v.words) &&
    isCount(v.letters) &&
    typeof v.longest === 'string' &&
    typeof v.first === 'string'
  );
}

function isLifetime(v: unknown): v is Lifetime {
  return (
    isRecord(v) &&
    (v.since === null || isDay(v.since)) &&
    isCount(v.daysPlayed) &&
    (v.lastDay === null || isDay(v.lastDay)) &&
    isCount(v.words) &&
    isCount(v.letters) &&
    typeof v.longestWord === 'string' &&
    (v.fewestTries === null || isCount(v.fewestTries)) &&
    isCounts(v.triesCounts, 7) &&
    isCount(v.noHintGrids) &&
    isRecord(v.firstGuesses) &&
    Object.values(v.firstGuesses).every(isCount) &&
    isCounts(v.hours, 24) &&
    isCount(v.mostInADay) &&
    isCount(v.todayCount)
  );
}

function read(store: KeyValueStore, key: string): unknown {
  try {
    const raw = store.getItem(key);
    return raw === null ? null : JSON.parse(raw);
  } catch {
    return null;
  }
}

function write(store: KeyValueStore, key: string, value: unknown): void {
  try {
    store.setItem(key, JSON.stringify(value));
  } catch {
    // Storage full or unavailable: the stats just stop growing; the games are unaffected.
  }
}

/** The detailed log, oldest first. Damaged entries are skipped. */
export function loadLog(store: KeyValueStore): PlayRecord[] {
  const value = read(store, LOG_KEY);
  return Array.isArray(value) ? value.filter(isPlayRecord) : [];
}

/** The lifetime figures; damaged ones start over. */
export function loadLifetime(store: KeyValueStore): Lifetime {
  const value = read(store, LIFETIME_KEY);
  return isLifetime(value) ? value : emptyLifetime();
}

/** The lifetime figures with one more finished puzzle or word. Records arrive in day order. */
export function addToLifetime(life: Lifetime, r: PlayRecord): Lifetime {
  const newDay = life.lastDay !== r.day;
  const todayCount = newDay ? 1 : life.todayCount + 1;
  const typedGuess = r.game === 'guess' && r.typed && r.tries > 0;
  const triesCounts = [...life.triesCounts];
  if (typedGuess) triesCounts[Math.min(r.tries, 7) - 1] += 1;
  return {
    since: life.since ?? r.day,
    daysPlayed: life.daysPlayed + (newDay ? 1 : 0),
    lastDay: r.day,
    words: life.words + r.words,
    letters: life.letters + r.letters,
    longestWord: r.longest.length > life.longestWord.length ? r.longest : life.longestWord,
    fewestTries: typedGuess ? Math.min(life.fewestTries ?? r.tries, r.tries) : life.fewestTries,
    triesCounts,
    noHintGrids: life.noHintGrids + (r.game === 'grid' && r.hints === 0 ? 1 : 0),
    firstGuesses: r.first ? { ...life.firstGuesses, [r.first]: (life.firstGuesses[r.first] ?? 0) + 1 } : life.firstGuesses,
    hours: life.hours.map((n, hour) => (hour === r.hour ? n + 1 : n)),
    mostInADay: Math.max(life.mostInADay, todayCount),
    todayCount,
  };
}

/** Saves a finished puzzle or word: appends it, drops detail older than KEEP_DAYS, and updates the lifetime figures. */
export function addPlay(store: KeyValueStore, record: PlayRecord): void {
  const cutoff = addDays(record.day, -KEEP_DAYS);
  const log = loadLog(store).filter((r) => r.day > cutoff);
  log.push(record);
  write(store, LOG_KEY, log);
  write(store, LIFETIME_KEY, addToLifetime(loadLifetime(store), record));
}

/** The level with the most records (ties go to the easier level); null when none has any. */
export function mostPlayedLevel(records: readonly PlayRecord[]): PlayLevel | null {
  let best: PlayLevel | null = null;
  let most = 0;
  for (const level of PLAY_LEVELS) {
    const n = records.filter((r) => r.level === level).length;
    if (n > most) {
      best = level;
      most = n;
    }
  }
  return best;
}
```

import { answerWords } from '../grid/puzzle';
import type { GridPuzzle } from '../grid/types';
import { dayKey } from '../shared/days';
import type { PlayLevel, PlayRecord } from './log';

/** How many words, their letters, and the first of the longest. */
export function wordFacts(words: readonly string[]): Pick<PlayRecord, 'words' | 'letters' | 'longest'> {
  let longest = '';
  for (const word of words) if (word.length > longest.length) longest = word;
  return { words: words.length, letters: words.reduce((n, word) => n + word.length, 0), longest };
}

const when = (now: Date): Pick<PlayRecord, 'day' | 'hour'> => ({ day: dayKey(now), hour: now.getHours() });

/** Word Circle: its answer words plus the extra words she found. Each Hint reveals one letter. */
export function circleRecord(p: {
  words: readonly string[];
  extras: readonly string[];
  revealed: Record<string, number[]>;
  level: PlayLevel;
  ms: number;
  now: Date;
}): PlayRecord {
  return {
    ...when(p.now),
    game: 'circle',
    level: p.level,
    ms: Math.round(p.ms),
    hints: Object.values(p.revealed).reduce((n, indices) => n + indices.length, 0),
    tries: 0,
    typed: true,
    daily: false,
    ...wordFacts([...p.words, ...p.extras]),
    first: '',
  };
}

/** Word Guess: one word, with her tries, Hint letters and first guess. */
export function guessRecord(p: {
  answer: string;
  guesses: readonly string[];
  hinted: readonly number[];
  typed: boolean;
  daily: boolean;
  ms: number;
  now: Date;
}): PlayRecord {
  return {
    ...when(p.now),
    game: 'guess',
    level: null,
    ms: Math.round(p.ms),
    hints: p.hinted.length,
    tries: p.guesses.length,
    typed: p.typed,
    daily: p.daily,
    ...wordFacts([p.answer]),
    first: p.guesses[0] ?? '',
  };
}

/** Word Grid: every word across and down in the answer. */
export function gridRecord(p: { puzzle: GridPuzzle; hinted: readonly number[]; level: PlayLevel; ms: number; now: Date }): PlayRecord {
  return {
    ...when(p.now),
    game: 'grid',
    level: p.level,
    ms: Math.round(p.ms),
    hints: p.hinted.length,
    tries: 0,
    typed: true,
    daily: false,
    ...wordFacts(answerWords(p.puzzle)),
    first: '',
  };
}

const count = (n: number): string => n.toLocaleString('en-US');
const plural = (n: number, one: string, many: string): string => `${count(n)} ${n === 1 ? one : many}`;
const average = (values: readonly number[]): number => values.reduce((a, b) => a + b, 0) / values.length;

function totalLabel(games: readonly StatsGame[], total: number): string {
  const words = games.some((g) => g.id === 'guess');
  const puzzles = games.some((g) => g.id !== 'guess');
  if (words && puzzles) return total === 1 ? 'puzzle or word' : 'puzzles and words';
  if (words) return total === 1 ? 'word' : 'words';
  return total === 1 ? 'puzzle' : 'puzzles';
}

/** True if a milestone (10, 25, 50 … 1,000, then every 500) lies after `from`, up to and including `to`. */
function passedMilestone(from: number, to: number): boolean {
  for (let n = from + 1; n <= to; n++) if (isMilestone(n)) return true;
  return false;
}

function totalCard({ games, streak, lifetime, today, seen }: HerStatsInput): HerStats['total'] {
  const value = games.reduce((n, g) => n + g.solved, 0);
  const lines: StatLine[] = [];
  const current = streak.lastPlayedDay === today || streak.lastPlayedDay === addDays(today, -1) ? streak.streakDays : 0;
  if (current >= 2) lines.push({ icon: 'sun', text: `${count(current)} days in a row`, isNew: false });
  if (lifetime.daysPlayed >= 1) lines.push({ icon: 'calendar', text: plural(lifetime.daysPlayed, 'day played', 'days played'), isNew: false });
  return { value, label: totalLabel(games, value), lines, isNew: seen !== null && passedMilestone(seen.total, value) };
}

function podiumCard({ games, log }: HerStatsInput): HerStats['podium'] {
  const ranked = [...games].sort((a, b) => b.solved - a.solved);
  const order = ranked.length === 3 ? [ranked[1], ranked[0], ranked[2]] : ranked.length === 2 ? [ranked[1], ranked[0]] : ranked;
  const top = ranked[0];
  const entries = order.map((g) => ({ id: g.id, title: g.title, solved: g.solved, top: g === top && g.solved > 0 }));
  const level = top && top.id !== 'guess' ? mostPlayedLevel(log.filter((r) => r.game === top.id)) : null;
  return { entries, note: level ? `Mostly on ${LEVEL_NAMES[level]}` : null };
}

/** Word Circle's most-played level, if her last 7 days there beat the 7 before (at least 3 puzzles in each). */
function quickerLevel(log: readonly PlayRecord[], today: string): PlayLevel | null {
  const recent = log.filter((r) => r.game === 'circle' && r.day >= addDays(today, -13) && r.day <= today);
  const level = mostPlayedLevel(recent);
  if (!level) return null;
  const atLevel = recent.filter((r) => r.level === level);
  const last = atLevel.filter((r) => r.day >= addDays(today, -6)).map((r) => r.ms);
  const before = atLevel.filter((r) => r.day < addDays(today, -6)).map((r) => r.ms);
  return last.length >= 3 && before.length >= 3 && average(last) < average(before) ? level : null;
}

function bestsCard({ games, lifetime, log, today, seen }: HerStatsInput): HerStats['bests'] {
  const has = (id: GameId) => games.some((g) => g.id === id);
  const circle = games.find((g) => g.id === 'circle');
  const fastest: StatLine[] = [];
  for (const level of PLAY_LEVELS) {
    const ms = circle?.bestMs?.[level];
    if (ms === undefined) continue;
    const before = seen?.circleBest[level];
    fastest.push({ icon: 'trophy', text: `${LEVEL_NAMES[level]} ${formatDuration(ms)}`, isNew: seen !== null && (before === undefined || ms < before) });
  }
  const others: StatLine[] = [];
  const quicker = circle ? quickerLevel(log, today) : null;
  if (quicker) others.push({ icon: 'rocket', text: `Getting quicker at ${LEVEL_NAMES[quicker]}!`, isNew: false });
  if (has('grid') && lifetime.noHintGrids >= 1) {
    others.push({ icon: 'target', text: `${plural(lifetime.noHintGrids, 'Word Grid', 'Word Grids')}, no hints`, isNew: false });
  }
  let tries: HerStats['bests']['tries'] = null;
  let triesNote: string | null = null;
  if (has('guess')) {
    const most = Math.max(...lifetime.triesCounts);
    if (most > 0) {
      const topIndex = lifetime.triesCounts.indexOf(most);
      tries = lifetime.triesCounts.map((n, i) => ({ label: TRIES_LABELS[i], count: n, top: i === topIndex }));
    } else {
      triesNote = 'Your Word Guess chart starts with your next word.';
    }
  }
  const fewestTriesIsNew =
    seen !== null && lifetime.fewestTries !== null && (seen.fewestTries === null || lifetime.fewestTries < seen.fewestTries);
  return { fastest, others, tries, triesNote, fewestTriesIsNew, isNew: fewestTriesIsNew || fastest.some((l) => l.isNew) };
}

function gardenCard({ log, today }: HerStatsInput): HerStats['garden'] {
  const perDay = new Map<string, number>();
  for (const r of log) perDay.set(r.day, (perDay.get(r.day) ?? 0) + 1);
  const flowers = Array.from({ length: GARDEN_DAYS }, (_, i) => {
    const n = perDay.get(addDays(today, i - (GARDEN_DAYS - 1))) ?? 0;
    return n === 0 ? 0 : n <= 2 ? 1 : n <= 5 ? 2 : 3;
  });
  const grown = flowers.filter((f) => f > 0).length;
  return { flowers, caption: grown === 0 ? "Your garden starts growing with today's puzzles" : `${plural(grown, 'flower', 'flowers')} in your garden` };
}

function partOfDay(hours: readonly number[]): string | null {
  if (hours.reduce((a, b) => a + b, 0) < 10) return null;
  let best: string | null = null;
  let most = 0;
  for (const [name, from, to] of PARTS) {
    const n = hours.slice(from, to + 1).reduce((a, b) => a + b, 0);
    if (n > most) {
      best = name;
      most = n;
    }
  }
  return best;
}

function dayOfYear(day: string): number {
  const [y, m, d] = day.split('-').map(Number);
  return Math.round((Date.UTC(y, m - 1, d) - Date.UTC(y, 0, 1)) / 86_400_000);
}

function factsCard({ lifetime, seen, today }: HerStatsInput): HerStats['facts'] {
  const word = lifetime.longestWord.length >= 5 ? lifetime.longestWord.toUpperCase() : null;
  const wordIsNew = word !== null && seen !== null && lifetime.longestWord.length > seen.longestWord.length;
  const candidates: string[] = [];
  if (lifetime.words >= 50) candidates.push(`You've found ${count(lifetime.words)} words`);
  if (lifetime.letters >= 200) candidates.push(`You've spelled ${count(lifetime.letters)} letters`);
  const [favorite, uses] = Object.entries(lifetime.firstGuesses).sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]))[0] ?? ['', 0];
  if (uses >= 3) candidates.push(`Your favorite first guess is ${favorite.toUpperCase()}`);
  const part = partOfDay(lifetime.hours);
  if (part) candidates.push(`You're ${part === 'morning' ? 'a' : 'an'} ${part} player`);
  if (lifetime.mostInADay >= 3) candidates.push(`Your most in one day: ${count(lifetime.mostInADay)}`);
  const fact = candidates.length > 0 ? candidates[dayOfYear(today) % candidates.length] : null;
  return { word, wordIsNew, fact, empty: word === null && fact === null ? 'Fun facts appear as you play.' : null };
}

export function herStats(input: HerStatsInput): HerStats {
  return { total: totalCard(input), podium: podiumCard(input), bests: bestsCard(input), garden: gardenCard(input), facts: factsCard(input) };
}

function snapshot({ games, lifetime }: HerStatsInput): Seen {
  return {
    total: games.reduce((n, g) => n + g.solved, 0),
    circleBest: { ...(games.find((g) => g.id === 'circle')?.bestMs ?? {}) },
    fewestTries: lifetime.fewestTries,
    longestWord: lifetime.longestWord,
  };
}

/** What she has seen after card `card` was shown: only what that card showed is updated (everything, the first time). */
export function seenAfter(seen: Seen | null, input: HerStatsInput, card: number): Seen {
  const now = snapshot(input);
  if (seen === null) return now;
  if (card === 0) return { ...seen, total: now.total };
  if (card === 2) return { ...seen, circleBest: now.circleBest, fewestTries: now.fewestTries };
  if (card === 4) return { ...seen, longestWord: now.longestWord };
  return seen;
}

export function loadSeen(store: KeyValueStore): Seen | null {
  try {
    const raw = store.getItem(SEEN_KEY);
    const v: unknown = raw === null ? null : JSON.parse(raw);
    if (typeof v !== 'object' || v === null) return null;
    const { total, circleBest, fewestTries, longestWord } = v as Record<string, unknown>;
    if (!Number.isInteger(total) || typeof longestWord !== 'string') return null;
    if (fewestTries !== null && !Number.isInteger(fewestTries)) return null;
    if (typeof circleBest !== 'object' || circleBest === null) return null;
    return { total: total as number, circleBest: circleBest as Seen['circleBest'], fewestTries: fewestTries as number | null, longestWord };
  } catch {
    return null;
  }
}

export function saveSeen(store: KeyValueStore, seen: Seen): void {

    store.setItem(SEEN_KEY, JSON.stringify(seen));
  } catch {
    // Storage unavailable: something new may be celebrated again next time.
  }
}

const average = (values: readonly number[]): number | null => (values.length === 0 ? null : values.reduce((a, b) => a + b, 0) / values.length);
const between = (r: PlayRecord, from: string, to: string): boolean => r.day >= from && r.day <= to;

/** A round number at or above `max` for the top of an axis: 1, 2 or 5 times a power of ten. */
export function niceMax(max: number): number {
  if (max <= 0) return 1;
  const power = 10 ** Math.floor(Math.log10(max));
  return [1, 2, 5, 10].map((m) => m * power).find((n) => n >= max) ?? 10 * power;
}

}

function table(
  log: readonly PlayRecord[],
  today: string,
  since: string | null,
  present: readonly GameId[],
  levels: ReadonlyMap<GameId, PlayLevel | null>,
): TableRow[] {
  const beforeFrom = addDays(today, -(2 * PERIOD_DAYS - 1));
  const last = log.filter((r) => between(r, addDays(today, -(PERIOD_DAYS - 1)), today));
  const before = log.filter((r) => between(r, beforeFrom, addDays(today, -PERIOD_DAYS)));
  const rows: TableRow[] = [];
  const row = (label: string, pick: (records: readonly PlayRecord[]) => number[], format: (v: number) => string) => {
    const a = average(pick(last));
    const b = average(pick(before));
    if (a !== null && b !== null) rows.push({ label, last: format(a), before: format(b) });
  };
  for (const game of present) {
    const level = levels.get(game) ?? null;
    const label = game === 'guess' || level === null ? `${GAME_TITLES[game]}: time per word` : `${GAME_TITLES[game]}, ${LEVEL_NAMES[level]}: time per puzzle`;
    row(label, (rs) => rs.filter((r) => r.game === game && r.level === level).map((r) => r.ms), formatDuration);
  }
  for (const game of present.filter((g) => g !== 'guess')) {
    row(`${GAME_TITLES[game]}: hints per puzzle`, (rs) => rs.filter((r) => r.game === game).map((r) => r.hints), (v) => v.toFixed(1));
  }
  if (present.includes('guess')) {
    row('Word Guess: tries per word', (rs) => rs.filter((r) => r.game === 'guess' && r.tries > 0).map((r) => r.tries), (v) => v.toFixed(1));
  }
  if (since !== null && since <= beforeFrom) {
    const played = (rs: readonly PlayRecord[]) => new Set(rs.map((r) => r.day)).size;
    rows.push({ label: 'Days played', last: `${played(last)} of ${PERIOD_DAYS}`, before: `${played(before)} of ${PERIOD_DAYS}` });
  }
  return rows;
}

/** The family's weekly figures from the play log. `since` is the first logged day (from the lifetime figures). */
export function familyStats(log: readonly PlayRecord[], today: string, since: string | null): FamilyStats {
  const weekStarts = Array.from({ length: WEEKS }, (_, i) => addDays(today, -(7 * (WEEKS - 1 - i) + 6)));
  const weekly = (records: readonly PlayRecord[], value: (r: PlayRecord) => number): (number | null)[] =>
    weekStarts.map((start) => average(records.filter((r) => between(r, start, addDays(start, 6))).map(value)));
  const recent = log.filter((r) => between(r, weekStarts[0], today));
  const present = GAME_IDS.filter((g) => recent.some((r) => r.game === g));
  const levels = new Map(present.map((g) => [g, g === 'guess' ? null : mostPlayedLevel(recent.filter((r) => r.game === g))] as const));

  const speed = present.map((g) => ({
    game: g,
    label: SHORT[g],
    points: weekly(recent.filter((r) => r.game === g && r.level === levels.get(g)), (r) => r.ms / 60_000),
  }));
  const hints = present.filter((g) => g !== 'guess').map((g) => ({ game: g, label: SHORT[g], points: weekly(recent.filter((r) => r.game === g), (r) => r.hints) }));
  const guessed = recent.filter((r) => r.game === 'guess' && r.tries > 0);
  const tries = guessed.length > 0 ? { game: 'guess' as const, label: SHORT.guess, points: weekly(guessed, (r) => r.tries) } : null;

  const days = Array.from({ length: PERIOD_DAYS }, (_, i) => addDays(today, i - (PERIOD_DAYS - 1)));
  const perDay = days.map((day) => log.filter((r) => r.day === day).length);
  const period = log.filter((r) => between(r, days[0], today));
  const parts = PARTS.map(([, from, to]) => period.filter((r) => r.hour >= from && r.hour <= to).length);
  const shares = [...parts, period.length - parts.reduce((a, b) => a + b, 0)];
  const timeOfDay = ['Morning', 'Afternoon', 'Evening', 'Night'].map((label, i) => ({
    label,
    share: period.length === 0 ? 0 : Math.round((100 * shares[i]) / period.length),
  }));
  const games = present.map((g) => ({
    game: g,
    label: GAME_TITLES[g],
    counts: weekStarts.map((start) => recent.filter((r) => r.game === g && between(r, start, addDays(start, 6))).length),
  }));

  return {
    since,
    ready: since !== null && since <= addDays(today, -(READY_DAYS - 1)),
    table: table(log, today, since, present, levels),
    speed,
    hints,
    tries,
    days,
    perDay,
    timeOfDay,
    games,
    weekStarts,
  };
}

function confetti(card: HTMLElement): void {

  }
}

  constructor(parent: HTMLElement) {
    this.root = el('div', 'stats-popup', parent);
  }

  /** Shows card `index` (0–4), animating it in; confetti if it has something new. */
  show(stats: HerStats, index: number, layout: FrameLayout): void {
    cancelAnimationFrame(this.countUp);
    this.root.replaceChildren();
    el('div', 'celebrate-dim', this.root);
    const card = el('div', 'stats-card', this.root);
    CARDS[index % CARD_COUNT](card, stats);
    const close = el('button', 'next close', this.root);
    close.type = 'button';
    close.innerHTML = `${CLOSE_ICON}<span>Close</span>`;
    const next = el('button', 'next forward', this.root);
    next.type = 'button';
    next.innerHTML = `<span>Next</span>${ARROW_ICON}`;
    this.root.classList.add('visible');
    this.place(layout);
    if (reducedMotion()) return;
    if (index === 0) this.countUpTo(card, stats.total.value);
    const isNew = [stats.total.isNew, false, stats.bests.isNew, false, stats.facts.wordIsNew][index];
    if (isNew) confetti(card);
  }

  place(layout: FrameLayout): void {
    const u = layout.card.h / 100;
    this.root.style.setProperty('--u', `${u}px`);
    const card = this.root.querySelector<HTMLElement>('.stats-card');
    if (card) place(card, layout.card);
    const tiles = this.root.querySelector<HTMLElement>('.stats-tiles');
    if (tiles) tiles.style.setProperty('--tile', `${Math.min(TILE_UNITS * u, (layout.card.w * 0.85) / tiles.childElementCount - 8)}px`);
    const close = this.root.querySelector<HTMLElement>('.close');
    if (close) place(close, layout.shareButton);
    const next = this.root.querySelector<HTMLElement>('.forward');
    if (next) place(next, layout.nextWordButton);
  }

  hide(): void {
    cancelAnimationFrame(this.countUp);
    this.root.classList.remove('visible');
    this.root.replaceChildren();
  }

  /** The total counts up from 0, easing out, over COUNT_UP_MS. */
  private countUpTo(card: HTMLElement, total: number): void {
    const node = card.querySelector<HTMLElement>('.stats-big');
    if (!node) return;
    const start = performance.now();
    const step = (now: number) => {
      const p = Math.min(1, (now - start) / COUNT_UP_MS);
      node.textContent = Math.round(total * (1 - (1 - p) ** 3)).toLocaleString('en-US');
      if (p < 1) this.countUp = requestAnimationFrame(step);
    };
    node.textContent = '0';
    this.countUp = requestAnimationFrame(step);
  }
}

import type { GameId } from '../frame/games';
import { el } from '../ui/dom';
import { niceMax, shortDay, type FamilyStats, type Series } from './family-stats';

/** Word Circle, Word Guess, Word Grid: the reference palette's first three slots, validated as a set. */
const COLORS: Record<GameId, string> = { circle: '#2a78d6', guess: '#eb6834', grid: '#1baf7a' };
const NS = 'http://www.w3.org/2000/svg';
/** The chart box in SVG units; the plot runs from L to R across and T to B down. */
const W = 600;
const H = 200;
const L = 48;
const R = 520;
const T = 12;
const B = 168;

interface Chart {
  plot: SVGSVGElement;
  /** Shows a tapped mark's exact value under the chart. */
  show(text: string): void;
}

function svg<K extends keyof SVGElementTagNameMap>(parent: Element, tag: K, attrs: Record<string, string | number>, text?: string): SVGElementTagNameMap[K] {
  const node = document.createElementNS(NS, tag);
  for (const [name, value] of Object.entries(attrs)) node.setAttribute(name, String(value));
  if (text !== undefined) node.textContent = text;
  parent.appendChild(node);
  return node;
}

function chart(parent: HTMLElement, heading: string, note: string): Chart {
  const box = el('div', 'family-chart', parent);
  el('h3', '', box).textContent = heading;
  el('p', 'settings-note', box).textContent = note;
  const plot = svg(box, 'svg', { viewBox: `0 0 ${W} ${H}`, role: 'img', 'aria-label': heading });
  const value = el('p', 'family-value', box);
  value.textContent = 'Tap a point or bar for its number.';
  return { plot, show: (text) => (value.textContent = text) };
}

function axes(plot: SVGSVGElement, max: number, label: (value: number) => string): void {
  for (const f of [0, 0.5, 1]) {
    const y = B - (B - T) * f;
    svg(plot, 'line', { x1: L, x2: R, y1: y, y2: y, class: 'family-grid' });
    svg(plot, 'text', { x: L - 6, y: y + 4, 'text-anchor': 'end', class: 'family-axis' }, label(max * f));
  }
}

function xLabels(plot: SVGSVGElement, first: string, last: string): void {
  svg(plot, 'text', { x: L, y: H - 8, class: 'family-axis' }, first);
  svg(plot, 'text', { x: R, y: H - 8, 'text-anchor': 'end', class: 'family-axis' }, last);
}

function lineChart(
  c: Chart,
  series: readonly Series[],
  weekStarts: readonly string[],
  axisLabel: (value: number) => string,
  tapText: (s: Series, value: number, weekStart: string) => string,
): void {
  const values = series.flatMap((s) => s.points.filter((p): p is number => p !== null));
  const max = niceMax(Math.max(0, ...values));
  axes(c.plot, max, axisLabel);
  const x = (i: number) => L + ((R - L) * i) / (weekStarts.length - 1);
  const y = (v: number) => B - ((B - T) * v) / max;
  for (const s of series) {
    let d = '';
    let drawing = false;
    s.points.forEach((p, i) => {
      if (p === null) {
        drawing = false;
        return;
      }
      d += `${drawing ? 'L' : 'M'}${x(i)} ${y(p)} `;
      drawing = true;
    });
    svg(c.plot, 'path', { d, fill: 'none', stroke: COLORS[s.game], 'stroke-width': 2.5, 'stroke-linejoin': 'round', 'stroke-linecap': 'round' });
    let last = -1;
    s.points.forEach((p, i) => {
      if (p === null) return;
      last = i;
      svg(c.plot, 'circle', { cx: x(i), cy: y(p), r: 4, fill: COLORS[s.game] });
      const hit = svg(c.plot, 'circle', { cx: x(i), cy: y(p), r: 14, class: 'family-hit' });
      hit.addEventListener('click', () => c.show(tapText(s, p, weekStarts[i])));
    });
    if (last >= 0) svg(c.plot, 'text', { x: x(last) + 8, y: y(s.points[last]!) + 5, class: 'family-label' }, s.label);
  }
  xLabels(c.plot, '12 weeks ago', 'This week');
}

function dayBars(c: Chart, perDay: readonly number[], days: readonly string[]): void {
  const max = niceMax(Math.max(0, ...perDay));
  axes(c.plot, max, (v) => String(Math.round(v)));
  const w = (R - L) / perDay.length;
  perDay.forEach((n, i) => {
    const h = ((B - T) * n) / max;
    if (n > 0) svg(c.plot, 'rect', { x: L + i * w + 1, y: B - h, width: w - 2, height: h, rx: 2, fill: COLORS.circle });
    const hit = svg(c.plot, 'rect', { x: L + i * w, y: T, width: w, height: B - T, class: 'family-hit' });
    hit.addEventListener('click', () => c.show(`${shortDay(days[i])}: ${n} ${n === 1 ? 'puzzle' : 'puzzles'}`));
  });
  xLabels(c.plot, '4 weeks ago', 'Today');
}

function partBars(c: Chart, parts: FamilyStats['timeOfDay']): void {
  parts.forEach((part, i) => {
    const y = T + 8 + i * 40;
    svg(c.plot, 'text', { x: 110, y: y + 19, 'text-anchor': 'end', class: 'family-label' }, part.label);
    const w = ((R - 130) * part.share) / 100;
    if (part.share > 0) svg(c.plot, 'rect', { x: 120, y, width: w, height: 26, rx: 4, fill: COLORS.circle });
    svg(c.plot, 'text', { x: 128 + w, y: y + 19, class: 'family-axis' }, `${part.share}%`);
  });
}

function gameBars(c: Chart, games: FamilyStats['games'], weekStarts: readonly string[]): void {
  const totals = weekStarts.map((_, i) => games.reduce((n, g) => n + g.counts[i], 0));
  const max = niceMax(Math.max(0, ...totals));
  axes(c.plot, max, (v) => String(Math.round(v)));
  const w = (R - L) / weekStarts.length;
  weekStarts.forEach((start, i) => {
    let base = B;
    for (const g of games) {
      const n = g.counts[i];
      if (n === 0) continue;
      const h = ((B - T) * n) / max;
      const bar = svg(c.plot, 'rect', { x: L + i * w + 3, y: base - h + 1, width: w - 6, height: Math.max(h - 2, 1), rx: 2, fill: COLORS[g.game] });
      bar.addEventListener('click', () => c.show(`${g.label}, week of ${shortDay(start)}: ${n}`));
      base -= h;
    }
  });
  games.forEach((g, i) => {
    svg(c.plot, 'rect', { x: R + 12, y: T + i * 22, width: 12, height: 12, rx: 3, fill: COLORS[g.game] });
    svg(c.plot, 'text', { x: R + 30, y: T + 11 + i * 22, class: 'family-axis' }, g.label.replace('Word ', ''));
  });
  xLabels(c.plot, '12 weeks ago', 'This week');
}

/** Minutes on an axis: "30s", "1.5m", "12m". */
function minutes(v: number): string {
  if (v < 1) return `${Math.round(v * 60)}s`;
  return `${v < 10 ? v.toFixed(1) : Math.round(v)}m`;
}

/** The family's honest weekly view: a comparison table, then the charts. */
export function renderFamily(parent: HTMLElement, stats: FamilyStats): void {
  if (!stats.ready) {
    el('p', 'settings-note', parent).textContent = stats.since
      ? `Charts appear after about 2 weeks of play (since ${shortDay(stats.since)}).`
      : 'Charts appear after about 2 weeks of play.';
    return;
  }
  if (stats.table.length > 0) {
    const table = el('table', 'family-table', parent);
    const head = el('tr', '', table);
    for (const text of ['', 'Last 4 weeks', '4 weeks before']) el('th', '', head).textContent = text;
    for (const row of stats.table) {
      const tr = el('tr', '', table);
      for (const text of [row.label, row.last, row.before]) el('td', '', tr).textContent = text;
    }
  }
  const week = (start: string) => `week of ${shortDay(start)}`;
  if (stats.speed.length > 0) {
    lineChart(chart(parent, 'Speed', 'Average active time per puzzle, each week, at the level played most'), stats.speed, stats.weekStarts, minutes, (s, v, start) =>
      `${s.label}, ${week(start)}: ${formatDuration(v * 60_000)} per puzzle`);
  }
  if (stats.hints.length > 0) {
    lineChart(chart(parent, 'Hints per puzzle', 'Letters revealed by Hint, each week'), stats.hints, stats.weekStarts, (v) => v.toFixed(1), (s, v, start) =>
      `${s.label}, ${week(start)}: ${v.toFixed(1)} hints per puzzle`);
  }
  if (stats.tries) {
    lineChart(chart(parent, 'Word Guess: tries per word', 'Lower means the word was found sooner'), [stats.tries], stats.weekStarts, (v) => v.toFixed(1), (_, v, start) =>
      `${week(start)}: ${v.toFixed(1)} tries per word`);
  }
  dayBars(chart(parent, 'Puzzles per day', 'The last 4 weeks'), stats.perDay, stats.days);
  partBars(chart(parent, 'Time of day', 'When puzzles were finished, the last 4 weeks'), stats.timeOfDay);
  if (stats.games.length > 0) gameBars(chart(parent, 'Games', 'Puzzles per week, by game'), stats.games, stats.weekStarts);
}
